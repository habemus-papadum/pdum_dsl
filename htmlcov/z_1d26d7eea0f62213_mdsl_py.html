1"""The marker DSL: composite markers and reducers as declared data. 

2 

3The Node schema (Arg / Const / Prim) lives in nodes.py — the declared 

4stability boundary. Bodies lower through the AST PRODUCER (producer.py): 

5one named, inspectable tree over primitives per marker — by inspection, 

6never execution. Declarations register into 

7the cache-backed registries (registry.py — idempotent, 

8derivation-under-cache, conflict-refusing); partial derivatives are 

9DERIVED by tree rewriting over the one derivative table (derivative.py), 

10so new activation functions differentiate automatically. 

11 

12Composite REDUCERS (`defreducer`) carry structured state — a count of 

13scalar slots, or a RECORD class (state=State: fields by attribute, laid 

14out in field order). This is the shape SSM / linear-recurrence scans and 

15the online-softmax accumulator need. Associativity is DECLARED, not 

16verified — property-test it now; it becomes a typeclass instance 

17obligation in the Lean model. 

18 

19No control flow in marker bodies: `where` is the branch (matching the 

20program IR's no-branching rule) — straight-line is detected at lowering. 

21Constants are VALUE-space, so floats are honest here (0.5, sqrt(2)/2); 

22Fraction stays available for exact rationals — the coordinates-exact / 

23values-inexact doctrine, applied. 

24"""

60def node_digest(node: Node) -> str: 

61 """Content address of a tree (dataclass reprs are deterministic). 

62 Digest-derived names make registration automatic and idempotent: the 

63 same lambda traced in a loop lands on the same registry entry, so the 

64 registry behaves as a cache rather than a namespace — the main repo's 

65 build-in-a-loop-against-a-cache philosophy, applied here.""" 

66 return hashlib.sha1(repr(node).encode()).hexdigest()[:10]

90 def partial(self, i: int) -> "CompositeMarker": 

91 """The i-th partial derivative — a cache entry computed on demand 

92 from a cache entry: the tree rewrite runs once per name, ever.""" 

93 if not 0 <= i < self.arity: 

94 raise IndexError(f"{self.name} has arity {self.arity}") 

95 name = f"{self.name}.d{i}" 

96 return MARKERS.derive(name, lambda: CompositeMarker(name, self.arity, diff(self.body, i)))

102@dataclass(frozen=True, eq=False) 

103class CompositeReducer: 

104 """Structured-state reducer: lift each element (m scalars) to a state 

105 (k scalars), combine states associatively, project the answer.""" 

106 

107 name: str 

108 state: int # k 

109 element: int # m 

110 lift: tuple # k Nodes over m Args 

111 combine: tuple # k Nodes over 2k Args (left state ++ right state) 

112 init: tuple # k numbers: the identity state (empty reduction) 

113 project: Node # 1 Node over k Args 

114 associative: bool = True # declared, not verified — property-test it 

115 

116 def __repr__(self) -> str: 

117 return self.name

121 def component_markers(self): 

122 """(C_i over 2k args, L_j over m args, P over k args) as composite 

123 markers, so their partials come from the same derived-`diff` 

124 machinery as every other composite.""" 

125 k, m = self.state, self.element 

126 cs = tuple(_register_marker(f"{self.name}.C{i}", 2 * k, self.combine[i]) for i in range(k)) 

127 ls = tuple(_register_marker(f"{self.name}.L{j}", m, self.lift[j]) for j in range(k)) 

128 p = _register_marker(f"{self.name}.P", k, self.project) 

129 return cs, ls, p

142 def adjoint_scanner(self, i: int) -> "CompositeReducer": 

143 """The reversed-time backward recurrence r_u = M_u·r_{u-1} + g_u as a 

144 matrix linear-recurrence reducer over state (M: k×k, v: k), 

145 projecting v_i. The M part of the state is never projected, so the 

146 first element's M slot (which has no recurrence edge to carry) may 

147 hold boundary garbage harmlessly.""" 

148 name = f"{self.name}.adj{i}" 

149 k = self.state 

150 s = k * k + k 

151 return REDUCERS.derive( 

152 name, 

153 lambda: CompositeReducer( 

154 name=name, 

155 state=s, 

156 element=s, 

157 lift=tuple(Arg(j) for j in range(s)), 

158 combine=_matrix_pair_nodes(k), 

159 init=tuple(1.0 if (j < k * k and j // k == j % k) else 0.0 for j in range(s)), 

160 project=Arg(k * k + i), 

161 ), 

162 )

165def _sum_nodes(terms): 

166 if not terms: 

167 return Const(0) 

168 total = terms[0] 

169 for t in terms[1:]: 

170 total = Prim("add", (total, t)) 

171 return total

174def _matrix_pair_nodes(k: int) -> tuple: 

175 """Combine for the matrix linear recurrence: (Ml,vl) ⊕ (Mr,vr) = 

176 (Mr·Ml, Mr·vl + vr). State layout: M row-major (k²), then v (k). 

177 Associative by matrix algebra — the generic backward-scan carrier.""" 

178 s = k * k + k 

179 

180 def ml(r, c): 

181 return Arg(r * k + c) 

182 

183 def vl(r): 

184 return Arg(k * k + r) 

185 

186 def mr(r, c): 

187 return Arg(s + r * k + c) 

188 

189 def vr(r): 

190 return Arg(s + k * k + r) 

191 

192 out = [] 

193 for r in range(k): 

194 for c in range(k): 

195 out.append(_sum_nodes([Prim("mul", (mr(r, x), ml(x, c))) for x in range(k)])) 

196 for r in range(k): 

197 out.append(Prim("add", (_sum_nodes([Prim("mul", (mr(r, x), vl(x))) for x in range(k)]), vr(r)))) 

198 return tuple(out)

201def defmarker(name: str | None, arity: int, fn) -> CompositeMarker: 

202 """Lower a plain function (a def or a lambda) into a composite marker 

203 via the AST producer — one named, inspectable body tree over primitives 

204 (the marker-granularity gate, S.2). Nothing is executed. 

205 

206 sigmoid = defmarker("sigmoid", 1, lambda x: 1 / (1 + exp(-x))) 

207 

208 `name=None` derives a content-addressed name (`m_<digest>`) from the 

209 lowered tree — naming becomes optional, and re-lowering the same body 

210 in a loop dedupes onto the same registry entry (the registry as cache, 

211 not namespace). 

212 """ 

213 if name in _PRIMITIVE_NAMES: 

214 raise ValueError(f"{name!r} is a primitive marker name") 

215 (body,) = producer.lower(fn, producer.scalars(arity)) 

216 if name is None: 

217 name = f"m_{node_digest(body)}" 

218 return _register_marker(name, arity, body)

221def defreducer( 

222 name: str, *, state, element: int, lift, combine, init, project=None, associative: bool = True 

223) -> CompositeReducer: 

224 """Define a structured-state reducer; bodies lower via the AST producer. 

225 

226 `state` is a count OR a record class (a frozen dataclass / NamedTuple): 

227 with `state=k`, `lift(e1..em) -> k-tuple`, `combine(left, right)` takes 

228 k-tuples (subscripted by literal index) and `project(s1..sk) -> scalar` 

229 (default: the first component). With `state=State`, lift returns 

230 `State(...)`, combine takes two State-typed arguments (fields by 

231 attribute) and returns one, project takes ONE State argument, and 

232 `init` may be a `State(...)` instance — record-typed reducer state 

233 (S.2); the state layout is the record's field order. 

234 

235 linrec = defreducer("linrec", state=2, element=2, 

236 lift=lambda a, b: (a, b), 

237 combine=lambda l, r: (l[0] * r[0], r[0] * l[1] + r[1]), 

238 init=(1.0, 0.0), 

239 project=lambda A, B: B) 

240 """ 

241 record = state if producer.is_record(state) else None 

242 k = len(producer.record_fields(record)) if record else state 

243 lift_nodes = producer.lower(lift, producer.scalars(element)) 

244 if len(lift_nodes) != k: 

245 raise ValueError(f"lift must produce {k} components, got {len(lift_nodes)}") 

246 if record: 

247 combine_nodes = producer.lower( 

248 combine, producer.record_binding(record, k) + producer.record_binding(record, k, k) 

249 ) 

250 project_bindings = producer.record_binding(record, k) 

251 else: 

252 combine_nodes = producer.lower(combine, producer.tuple_binding(k) + producer.tuple_binding(k, k)) 

253 project_bindings = producer.scalars(k) 

254 if len(combine_nodes) != k: 

255 raise ValueError(f"combine must produce {k} components") 

256 if producer.is_record(type(init)): 

257 init = tuple(getattr(init, f) for f in producer.record_fields(type(init))) 

258 if len(init) != k: 

259 raise ValueError(f"init must have {k} components") 

260 project_node = producer.lower(project, project_bindings)[0] if project is not None else Arg(0) 

261 r = CompositeReducer(name, k, element, lift_nodes, combine_nodes, tuple(init), project_node, associative) 

262 return REDUCERS.register( 

263 name, r, lambda v: (v.state, v.element, v.lift, v.combine, v.init, v.project, v.associative) 

264 )
