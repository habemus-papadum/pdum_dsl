1"""The tl dialect over the dsl core (240 C4.1) — promoted from the C3 spike. 

2 

3ONE IR, many dialects (the pivot's objective): tensor-typed SSA lives on 

4the dsl ``Node``/``Region`` infrastructure as the ``tl.*`` op family, with 

5the layout lattice AS the type — so tl's alignment law is an ordinary 

6type rule that refuses at emission with source locations, and content 

7keys (``region.key``) come from the core for free. 

8 

9Design items resolved here (the spike's frictions): 

10 

11- **The per-kind yield protocol.** Every region ends in ``core.yield`` 

12 (Julia's rule, at the IR level); WHAT a body yields is the KIND's 

13 declaration, never user ceremony: a ``step`` yields its returned 

14 value, a ``compute`` kernel yields the final ordering token (its 

15 effect) — and a kernel ``return`` refuses with the pinned voice. 

16- **Object-based intrinsic recognition.** ``thread_idx`` and the S.1 

17 vocabulary (``pointwise``, ``const_like``) are recognized by OBJECT 

18 IDENTITY from the function's own globals — never by name string. 

19- **The op-selection pattern, written once.** Operator/comparison rules 

20 are built by ONE factory (``_typed_rule``): lower the children, and 

21 TENSOR-TYPEDNESS selects the dialect path (a table row), else the 

22 base value pack serves the scalar subtree unchanged. A dialect 

23 extends by rows, not by wrapping rules. 

24- **The two-pass mechanism (owner-ruled).** Pass 1 

25 (``check_fold_step_supported``) walks a region and refuses anything 

26 the machinery cannot handle, WITH the reason; pass 2 does the work. 

27 

28The labeling frame (charts/labels/levels) is IN type identity — the 

29degenerate frame stays implicit, per the chart doctrine — so frames 

30misalign at emission and content keys distinguish them; dtype stays at 

31the compute layer (tensor.py's doctrine) until a backend column asks 

32(C5+). Schedules (store-all vs revolve) are EVALUATION STRATEGIES over 

33the same regions — never IR (the C3b finding); an L4 certified descent 

34may bake one later.

219def _r_fold(args, attrs, regions): 

220 """The widened fold contract (the excavation, LEVELS — the carry-dict 

221 debt burned): args = k state inits ++ m element sources; the step's 

222 binders match positionally; the step YIELDS the k next states in state 

223 order (+ the emitted value iff out=("emit",)); out=("final", i) returns 

224 state i's final value. The name tuples are contract (reporting keys); 

225 the carry is POSITIONAL — no name-based carry map exists.""" 

226 at = dict(attrs) 

227 state, element = tuple(at["state"]), tuple(at["element"]) 

228 k, m = len(state), len(element) 

229 if k < 1: 

230 raise TypeError("tl.fold needs at least one state tensor") 

231 if len(args) != k + m: 

232 raise TypeError(f"tl.fold takes {k} state init(s) + {m} element source(s), got {len(args)} operands") 

233 if not all(isinstance(a, TensorType) for a in args): 

234 raise TypeError("tl.fold wants tensor-typed state inits and element sources") 

235 (step,) = regions 

236 dim = at["dim"] 

237 if m == 0 and at.get("extent") is None: 

238 raise TypeError("tl.fold without element sources needs extent=(start, stop)") 

239 inits, srcs = args[:k], args[k:] 

240 for s in srcs: 

241 d = next((dd for dd in s.dims if dd.name == dim), None) 

242 if d is None: 

243 raise TypeError(f"tl.fold: element source has no scan dim {dim!r}") 

244 if d.chart is not None or d.labels is not None: 

245 raise TypeError(f"fold scan dim {dim!r} must be chartless/unlabeled (strip_charts first)") 

246 elems = tuple(_minus_dim(s, dim) for s in srcs) 

247 binders = tuple(p.type for p in step.params) 

248 want = tuple(inits) + elems

250 def _frames(tt): 

251 # order-insensitive frame agreement: presentation order is never 

252 # semantics (220 §11); binders bind by position, frames by name 

253 return sorted(tt.dims, key=lambda d: d.name) 

254 

255 if len(binders) != len(want) or any(_frames(bb) != _frames(w) for bb, w in zip(binders, want)): 

256 raise TypeError(f"tl.fold: step binders {binders!r} do not match (state inits + element slices) {want!r}") 

257 yielded = step.body[-1].args[0] 

258 ytypes = tuple(a.type for a in yielded.args) if yielded.op == "core.tuple" else (yielded.type,) 

259 out = tuple(at["out"]) 

260 # the step declares k next-states (state order) + at most ONE emit slot; 

261 # out=("final", i) may run a step that also declares the slot (the 

262 # adjoint machinery re-outs one step region under several fold nodes) 

263 if len(ytypes) not in (k, k + 1): 

264 raise TypeError( 

265 f"tl.fold: the step must yield the {k} next state(s) in state order " 

266 f"(+ at most one emitted value) — got {len(ytypes)}" 

267 ) 

268 if out[0] == "emit" and len(ytypes) != k + 1: 

269 raise TypeError("tl.fold: out=('emit',) needs the step to yield the emitted value after the states") 

270 for i in range(k): 

271 if _frames(ytypes[i]) != _frames(inits[i]): 

272 raise TypeError(f"tl.fold: carry {state[i]!r} changes the state type: {inits[i]!r} -> {ytypes[i]!r}") 

273 if out[0] == "final": 

274 return inits[out[1]] 

275 start, stop = _src_domain(srcs[0], dim) if m else tuple(at["extent"]) 

276 return _of_layout(_dense_like((Dim(dim, 0, start, stop),) + tuple(ytypes[k].layout.dims)))

303def _r_bridge(base): 

304 """The migration bridge (240 C4.3): the type rule IS the incumbent 

305 shadow inference — build the instruction, ask ``infer_instr``. One 

306 source of truth for layout semantics while both worlds coexist.""" 

307 

308 def rule(args, attrs, regions): 

309 names = tuple(f"a{i}" for i in range(len(args))) 

310 shadows = {} 

311 for n, t in zip(names, args): 

312 if not isinstance(t, TensorType): 

313 raise TypeError(f"tl.{base} wants tensor operands, got {t!r}") 

314 shadows[n] = t.layout 

315 return _of_layout(_infer_base(base, [shadows[n] for n in names], _thaw_params(dict(attrs)))) 

316 

317 return rule

349def _infer_base(base: str, lays, p): 

350 """One op's shadow from its operands' layouts (the retired infer_instr's 

351 rows). Layouts only — no data, no names.""" 

352 from .guarded import GuardedLayout, pad_layout, stencil_layout 

353 

354 if base == "const": 

355 dims = tuple( 

356 Dim(name, 0, *(extent if isinstance(extent, tuple) else (0, extent))) for name, extent in p.get("dims", ()) 

357 ) 

358 return Layout(dims) # stride-0 broadcast, exactly like the eval row 

359 if base == "random": 

360 return _dense_like(lays[0].dims) 

361 if base == "reduce": 

362 dims = p["dims"] 

363 dim_names = (dims,) if isinstance(dims, str) else tuple(dims) 

364 return _dense_like(tuple(d for d in lays[0].dims if d.name not in dim_names)) 

365 if base == "scan": 

366 return _dense_like(lays[0].dims) 

367 if base == "materialize": 

368 from dataclasses import replace as _replace 

369 

370 dims = tuple(_replace(lays[0].dim(n), chart=None, labels=None) for n in tuple(p["order"])) 

371 return _dense_like(dims) 

372 if base == "stage": 

373 # 320 §4: the ONE copying op with a residence — a dense copy in the 

374 # declared order at the named level. Charts RIDE (same physics, 

375 # different residence); chart-stripping stays materialize's separate 

376 # contract. `level` is a name resolved by cost analysis, never here. 

377 order = p.get("order") 

378 dims = tuple(lays[0].dim(n) for n in tuple(order)) if order else lays[0].dims 

379 return _dense_like(dims) 

380 if base == "round_to": 

381 return _dense_like(lays[0].dims) 

382 if base == "repeat_like": 

383 # the batching-unawareness mechanism (220): added dims are LAYOUT- 

384 # DERIVED from the like operand — referenced for its layout only 

385 x, like = lays[0], lays[1] 

386 have = {d.name for d in x.dims} 

387 for d in like.dims: 

388 if d.name not in have: 

389 x = x.repeat(d.name, (d.start, d.stop), d.chart, d.labels) 

390 if d.level is not None: 

391 x = x.bind(**{d.name: d.level}) 

392 return x 

393 if base == "take": 

394 from .indexing import infer_take 

395 

396 return infer_take(lays, p) 

397 if base == "scatter_add": 

398 from .indexing import infer_scatter 

399 

400 return infer_scatter(lays, p) 

401 if base in ("argtopk", "argsort"): 

402 from .indexing import infer_producer

404 return infer_producer(base, lays, p) 

405 if base == "pad": 

406 return pad_layout(lays[0], p["extents"]) 

407 if base == "stencil": 

408 return stencil_layout(lays[0], p["name"], p["k"], p.get("k_name"), p.get("dilation", 1)) 

409 if base == "simplify": 

410 s = lays[0] 

411 return s.simplify() if isinstance(s, GuardedLayout) else s 

412 if base == "with_value_units": 

413 return lays[0] 

414 return LAYOUT_ADAPTERS[base](lays[0], p)

534 ]

543 o if hasattr(o, "type") else ctx.emit("core.const", node=node, type=f64, value=float(o))

671def _tl_call(ctx, node): 

672 """Call resolution, OBJECT-IDENTITY-FIRST: the tl intrinsics and the S.1 

673 vocabulary are recognized as the objects the body's globals actually 

674 hold (direct-name imports — the package attr ``compute`` is the 

675 DECORATOR, the P5 shadowing lesson); markers via the registry; 

676 everything else is the base pack.""" 

677 if isinstance(node.func, pyast.Name): 

678 obj = _lookup(ctx, node.func.id) 

679 if isinstance(obj, _Intrinsic) and obj.name == "thread_idx": 

680 lattice = ctx.root.params[-1] # the writable target (S.3 convention) 

681 names = [c.value for c in node.args] 

682 out = [] 

683 for n in names: # Coordinates (250): the iota backing, frame-typed 

684 backing = ctx.emit("tl.iota", lattice, node=node, name=n) 

685 frame = next(f for f in lattice.type.dims if f.name == n) 

686 out.append(ctx.emit("tl.coord", backing, node=node, frame=frame))

688 if isinstance(obj, _Intrinsic) and obj.name in ("f32", "i32"): 

689 return _coerce_scalar(obj.name, ctx.lower(node.args[0])) 

690 if obj is _eager_pw: # the S.1 STEP-tier spelling 

691 marker = _lookup(ctx, node.args[0].id) or ctx.context["registry"].overloads.get(node.args[0].id) 

692 if marker is None or not hasattr(marker, "name"): 

693 raise ValueError(f"pointwise wants a marker first, got {node.args[0].id!r}") 

694 rest = [_scalar_lift(ctx, node, ctx.lower(a)) for a in node.args[1:]] 

695 return ctx.emit("tl.pointwise", *rest, node=node, f=marker.name) 

696 if obj is const_like: # scalar broadcast: the const IS the operand 

697 try: 

698 v = _host(ctx, node.args[1]) 

699 except Exception: 

700 v = None 

701 if isinstance(v, int) and not isinstance(v, bool): 

702 # the literal's own type is the carrier declaration (the 

703 # eager face's law): an int MATERIALIZES at integer carrier 

704 # over the ref's lattice — index arithmetic (§1.9 

705 # linearizations) never rides f64. Floats stay the DEFERRED 

706 # scalar (pointwise's broadcast law; fold steps require it). 

707 ref = ctx.lower(node.args[0]) 

708 dims = tuple((d.name, (d.start, d.stop)) for d in ref.type.layout.dims) 

709 return ctx.emit("tl.const", node=node, value=v, dims=dims, dtype="int64") 

710 return ctx.lower(node.args[1]) 

711 if obj is _eager_reduce or obj is _eager_scan: # the S.1 spellings 

712 f = _host(ctx, node.args[0]) 

713 fname = f if isinstance(f, str) else f.name 

714 arg1 = node.args[1] 

715 if isinstance(arg1, pyast.Tuple): 

716 operands = tuple(ctx.lower(e) for e in arg1.elts) 

717 else: 

718 operand = ctx.lower(arg1) 

719 operands = operand if isinstance(operand, tuple) else (operand,) 

720 dims = _host(ctx, node.args[2]) if len(node.args) > 2 else _host(ctx, node.keywords[0].value) 

721 key = "dims" if obj is _eager_reduce else "dim" 

722 op = "tl.reduce" if obj is _eager_reduce else "tl.scan" 

723 return ctx.emit(op, *operands, node=node, f=fname, **{key: dims}) 

724 if obj is _eager_repeat_like: # THE alignment primitive (S.1) 

725 x, like = (ctx.lower(a) for a in node.args) 

726 return ctx.emit("tl.repeat_like", x, like, node=node) 

727 if obj is _eager_iota: 

728 src = ctx.lower(node.args[0]) 

729 extra = {"unit": _host(ctx, node.args[2])} if len(node.args) > 2 else {} 

730 return ctx.emit("tl.iota", src, node=node, name=_host(ctx, node.args[1]), **extra) 

731 if obj is _eager_extent: # a structural READ: host data from the TYPE 

732 t = ctx.lower(node.args[0]) 

733 if hasattr(t, "type") and isinstance(t.type, CoordType): 

734 return t.type.frame.size # the coordinate face: the frame's width, a host INT 

735 want = _host(ctx, node.args[1])

738 name = obj.__name__ 

739 arity = 2 if obj in (_eager_take, _eager_scatter_add) else 1

741 raise TypeError(f"{name} takes {arity} tensor operand(s) — dim/extent/k/k_name are keywords") 

742 ops = [ctx.lower(a) for a in node.args] 

743 kw = {k.arg: _host(ctx, k.value) for k in node.keywords} 

744 if name == "scatter_add" and not hasattr(ops[0], "type"): 

745 # a scalar values operand (a deferred const_like float) 

746 # materializes over the IDX lattice: one contribution per 

747 # index element — the count-scatter

750 ops[0] = ctx.emit("tl.const", node=node, value=float(ops[0]), dims=dims)

752 if obj is _eager_contract: # ONE visible line over the primitives (S.1) 

753 a, bb = (ctx.lower(x) for x in node.args) 

754 axis = _host(ctx, node.keywords[0].value) if node.keywords else _host(ctx, node.args[2]) 

755 names = (axis,) if isinstance(axis, str) else tuple(axis) 

756 ra = ctx.emit("tl.repeat_like", a, bb, node=node) 

757 rb = ctx.emit("tl.repeat_like", bb, a, node=node) 

758 prod = ctx.emit("tl.pointwise", ra, rb, node=node, f="mul") 

759 return ctx.emit("tl.reduce", prod, node=node, f="sum", dims=names) 

760 obj2 = obj if obj is not None else ctx.context["registry"].overloads.get(node.func.id) 

761 if isinstance(obj2, Marker) or type(obj2).__name__ == "CompositeMarker":

764 raise TypeError(_COORD_MATH) 

765 if any(hasattr(a, "type") and isinstance(a.type, TensorType) for a in args): 

766 if ctx.context.get("tl.kind") not in KERNEL_KINDS: # the tensor tiers spell pointwise 

767 raise ValueError( 

768 f"{obj2.name} is a marker — tensor-tier application is spelled " 

769 f"pointwise({obj2.name}, ...) (bare names lower only inside " 

770 f"scalar marker bodies)"

773 if callable(obj) and not isinstance(obj, (Marker, _Intrinsic)):

775 h_args = [_host(ctx, a) for a in node.args] 

776 h_kwargs = _host_kwargs(ctx, node.keywords)

783 raise ValueError( 

784 f"staged transform {node.func.id!r} returned " 

785 f"{type(result).__name__!r}, not a function citizen — staged "

787 )

791 if getattr(result, "fp", None) is not None: 

792 raise ValueError( 

793 f"{node.func.id!r} returned a function value at lower time without " 

794 f"being a declared staged transform — decorate it with @staged "

799 if getattr(obj, "__module__", "") == "pdum.tl.compute":

802 f"recognition, not inlining; the recognized set is the primitive set"

805 kwargs_l = {}

808 kwargs_l.update(_host(ctx, kw.value)) 

809 else: 

810 kwargs_l[kw.arg] = _arg_value(ctx, kw.value) 

811 return _inline_plain(ctx, obj, args_l, node, kwargs_l) 

812 if isinstance(node.func, pyast.Attribute): # the frozen layout-method family 

813 base = ctx.lower(node.func.value) 

814 name = node.func.attr 

815 if hasattr(base, "type") and isinstance(base.type, TensorType):

840def _inline_plain(ctx, fn, args, node, kwargs=None): 

841 """Capture-and-call: a plain helper inlines through a CHILD lowerer over 

842 the same rules and the same build context — its bindings claim, and its 

843 claims collide honestly (the naming law reaches inlined bodies). 

844 Keyword-only parameters bind from the call's kwargs, then defaults.""" 

845 handle = capture_shim(fn) 

846 check_coherence(handle) 

847 child = Lowerer(handle, ctx.rules, ctx.ops, ctx.derived, wrap=ctx.loc(node), context=ctx.context, root=ctx.root) 

848 code = fn.__code__ 

849 pos = list(code.co_varnames[: code.co_argcount]) 

850 kwonly = list(code.co_varnames[code.co_argcount : code.co_argcount + code.co_kwonlyargcount]) 

851 kwargs = dict(kwargs or {}) 

852 if len(args) > len(pos): 

853 raise TypeError(f"{fn.__qualname__} takes {len(pos)} positional arguments, got {len(args)}") 

854 child.locals.update(zip(pos, args)) 

855 for name in pos[len(args) :]: 

856 if name in kwargs: 

857 child.locals[name] = kwargs.pop(name) 

858 elif fn.__defaults__ and name in pos[len(pos) - len(fn.__defaults__) :]: 

859 child.locals[name] = fn.__defaults__[pos.index(name) - (len(pos) - len(fn.__defaults__))] 

860 else: 

861 raise TypeError(f"{fn.__qualname__} missing argument {name!r}") 

862 for name in kwonly: 

863 if name in kwargs: 

864 child.locals[name] = kwargs.pop(name) 

865 elif fn.__kwdefaults__ and name in fn.__kwdefaults__: 

866 child.locals[name] = fn.__kwdefaults__[name] 

867 else: 

868 raise TypeError(f"{fn.__qualname__} missing keyword-only argument {name!r}") 

869 if kwargs: 

870 raise TypeError(f"{fn.__qualname__} got unexpected keyword arguments {sorted(kwargs)}") 

871 return child.run_body()

1235def _pullback(b, order: list, out, seed) -> dict: 

1236 """Reverse accumulation over a substituted DAG — the ONE region adjoint 

1237 walker (240 C4.3b). Per-op rules: pointwise slopes spliced from THE 

1238 table; reduce sum/mean (repeat back, mean divides by the static reduced 

1239 numel); repeat_like (reduce-sum over the added dims — the like operand 

1240 is layout-reference only, per doctrine). Returns id(node) -> adjoint; 

1241 absence is an exact zero.""" 

1242 adj: dict[int, object] = {id(out): seed} 

1243 

1244 def acc(operand, term): 

1245 prev = adj.get(id(operand)) 

1246 adj[id(operand)] = term if prev is None else b.emit("tl.pointwise", prev, term, f="add") 

1247 

1248 for node in reversed(order): 

1249 a = adj.get(id(node)) 

1250 if a is None: 

1251 continue 

1252 if node.op == "tl.pointwise": 

1253 rules = TABLE[dict(node.attrs)["f"]] 

1254 for rule, operand in zip(rules, node.args): 

1255 if rule is None or operand.op == "core.const": 

1256 continue 

1257 acc(operand, b.emit("tl.pointwise", _splice_tl(b, rule(*node.args)), a, f="mul")) 

1258 elif node.op == "tl.reduce": 

1259 f = dict(node.attrs)["f"] 

1260 (operand,) = node.args 

1261 rep = b.emit("tl.repeat_like", a, operand) 

1262 if f == "mean": 

1263 red = {d.name for d in operand.type.dims} - {d.name for d in node.type.dims} 

1264 n_red = 1 

1265 for d in operand.type.dims: 

1266 if d.name in red: 

1267 n_red *= d.size 

1268 rep = b.emit("tl.pointwise", rep, b.emit("core.const", type=f64, value=float(n_red)), f="div") 

1269 acc(operand, rep) 

1270 elif node.op == "tl.repeat_like": 

1271 x = node.args[0] 

1272 added = tuple(d.name for d in node.type.dims if d.name not in {q.name for q in x.type.dims}) 

1273 acc(x, b.emit("tl.reduce", a, f="sum", dims=added)) 

1274 return adj

1479def fold_grad(step: Region, vjp: Region, init, src, dim: str, *, slots: int | None = None): 

1480 """d(sum(final state))/d(init) — store-all when ``slots`` is None, else 

1481 segment-checkpoint revolve: recompute each segment from its checkpoint 

1482 during the backward, RE-SELECTING elements at absolute coordinates 

1483 (the recompute theorem's mechanism; bit-identical by construction).""" 

1484 src_dims = [(d.name, d.start, d.stop) for d in src.layout.dims] 

1485 lo, hi = next((s, e) for (name, s, e) in src_dims if name == dim) 

1486 stride = hi - lo if slots is None else -(-(hi - lo) // slots) 

1487 checkpoints = {lo: init} 

1488 s = init 

1489 for q in range(lo, hi): 

1490 if q != lo and (q - lo) % stride == 0: 

1491 checkpoints[q] = s 

1492 s = run_region(step, [s, src.select(**{dim: q})]) 

1493 ds = const_like(s, 1.0) 

1494 starts = sorted(checkpoints) 

1495 for seg in reversed(range(len(starts))): 

1496 base = starts[seg] 

1497 end = starts[seg + 1] if seg + 1 < len(starts) else hi 

1498 states = [checkpoints[base]] 

1499 for q in range(base, end): # the recomputation the theorem is about 

1500 states.append(run_region(step, [states[-1], src.select(**{dim: q})])) 

1501 for q in reversed(range(base, end)): 

1502 ds = run_region(vjp, [states[q - base], src.select(**{dim: q}), ds]) 

1503 return ds

1538def region_names(region: Region, param_names: tuple, names_of: dict | None = None) -> dict[int, str]: 

1539 """The naming law over a region: params CLAIM their declared names; every 

1540 other node DERIVES from its binding name (``names_of``, first-binding-wins) 

1541 or its op's base name. This is the ONE assignment — ``export_program`` 

1542 renders it, and the region-native analyses key their reports by it, so 

1543 name-keyed reports agree across the migration view by construction. 

1544 Returns ``id(node) -> name`` (consts, yields, and the yielded tuple are 

1545 plumbing: unnamed).""" 

1546 from pdum.dsl.naming import Namer 

1547 

1548 names = Namer() 

1549 names_of = names_of or {} 

1550 out: dict[int, str] = {} 

1551 for pname, p in zip(param_names, region.params): 

1552 out[id(p)] = names.claim(pname) 

1553 yielded = region.body[-1].args[0] if region.body else None 

1554 for n in walk_region(region): 

1555 if id(n) in out or n.op in ("core.yield", "core.const") or (n.op == "core.tuple" and n is yielded): 

1556 continue
