1"""Static ops counting over the IR — exact scalar-operation tallies. 

2 

3Design position on the mul/add/MAC question: COUNT primitives by NAME and 

4never collapse to a single "flops" scalar. A Counter keyed by marker name 

5("mul": 12, "add": 9, "exp": 4, "copy": 8) is the exact, layout-derived 

6fact; what a mul, an add, or an exp COSTS is a property of a machine, so it 

7belongs in a cost model (a weights dict applied at the end), not in the 

8count. In particular exp is just the bucket "exp" — whether it is 1 unit, 

9~20 flops of polynomial, or a table lookup is the cost model's opinion. 

10 

11MACs are not a primitive either — they are a FUSION the hardware performs on 

12the contraction pattern. `fuse_mac=True` recognizes exactly that pattern in 

13the IR (a pointwise mul consumed solely by a reduce-sum) and reclassifies: 

14the muls become "mac" and the reduce's adds are absorbed (the accumulator 

15runs inside the MAC chain, starting at the identity) — so a matmul counts 

16m·n·k macs, matching the standard figure. Anything fancier (fusing adds 

17into FMA chains inside arbitrary expressions) is a compiler decision and is 

18deliberately not guessed at here. 

19 

20Counts follow the REFERENCE semantics, sizes from `ir.infer` shadows: 

21- pointwise: (tree ops of the marker; a primitive is one op) x output numel. 

22- reduce: (numel_in - numel_out) combine ops (fold cost, identity-seeded); 

23 mean adds numel_out divides. Composite reducers: lift ops x numel_in, 

24 combine ops x (numel_in - lines), project ops x lines (reduce) or 

25 x numel_in (scan), where lines = numel_in / dim size. 

26- scan: (numel_in - lines) combine ops. 

27- materialize: numel "copy" (the one moving op — counted in its own bucket, 

28 never conflated with arithmetic). 

29- take: numel_out in the "take" bucket — one read+write per output element 

30 (200 §1.9), in its OWN bucket, never "copy": random access vs sequential 

31 copy is a machine property the cost model prices, so the count keeps them 

32 apart. scatter_add: numel_values "scatter" (the movement) + numel_values 

33 "add" (the accumulate) — the dual: one read+add+write per input element. 

34- layout ops, iota, const, metadata: zero. Guarded operands count over the 

35 guard BOX (the reference layer evaluates fills too) — a λ-proportional 

36 refinement can come later; no silent narrowing here. 

37"""

39from __future__ import annotations

41from collections import Counter 

42from dataclasses import dataclass 

43from typing import Mapping 

44 

45from .nodes import Prim 

46from .registry import MARKERS, REDUCERS 

47 

48_RED_COMBINE = {"sum": "add", "prod": "mul", "max": "maximum", "min": "minimum", "mean": "add"} 

49_FREE_OPS = ("iota", "const", "random", "round_to", "repeat_like", "with_value_units")

52def _tree_ops(node) -> Counter: 

53 c: Counter = Counter() 

54 if isinstance(node, Prim): 

55 c[node.op] += 1 

56 for a in node.args: 

57 c += _tree_ops(a) 

58 return c 

59 

60 

61def _marker_ops(name: str) -> Counter: 

62 if name in MARKERS: 

63 return _tree_ops(MARKERS[name].body) 

64 return Counter({name: 1}) 

65 

66 

67def _numel(layout) -> int: 

68 n = 1 

69 for d in layout.dims: 

70 n *= d.size 

71 return n 

72 

73 

74def _scale(ops: Counter, n: int) -> Counter: 

75 return Counter({k: v * n for k, v in ops.items()}) if n > 0 else Counter() 

76 

77 

78@dataclass(frozen=True) 

79class ProgramOps: 

80 per_var: Mapping[str, Counter] 

81 total: Counter 

82 

83 def weighted(self, weights: Mapping[str, float] | None = None, default: float = 1.0) -> float: 

84 """Collapse the named buckets under a cost model. `weights` maps op 

85 names to costs (e.g. {"exp": 20, "div": 4, "mac": 1, "copy": 0}); 

86 unnamed ops cost `default`.""" 

87 weights = weights or {} 

88 return float(sum(v * weights.get(k, default) for k, v in self.total.items()))

91def ops_count(region, input_layouts: dict | None = None, fuse_mac: bool = False, *, names=None) -> ProgramOps: 

92 """Count scalar operations per named value (and in total), by name. 

93 Sizes ride the region nodes' layout shadows (``input_layouts`` is not 

94 consulted; the argument survives for call-site symmetry with the other 

95 analyses); reports key by the naming law — pass ``names``.""" 

96 return _region_ops_count(region, fuse_mac, names)

99def _region_ops_count(region, fuse_mac: bool, names) -> ProgramOps: 

100 """The Region face — the same tallies over dialect nodes. Sizes come from 

101 the nodes' layout shadows; consts are deferred scalars (zero-cost 

102 plumbing, no report row); fuse_mac's sole-consumer test is DAG 

103 in-degree, with yields and the yielded tuple never counting as 

104 consumers (outputs are not consumption, matching the Program face).""" 

105 from .dialect import LAYOUT_OP_NAMES, region_names, walk_region # lazy: dialect imports this module's consumers 

106 

107 if names is None: 

108 raise ValueError( 

109 "region ops_count joins on names — pass names=region_names(region, param_names) " 

110 "(LiftedStep.names / Assemblage.names carry it)" 

111 ) 

112 per: dict[str, Counter] = {} 

113 for p in region.params: 

114 per[names[id(p)]] = Counter() 

115 yielded = region.body[-1].args[0] if region.body else None 

116 seen: set[int] = {id(p) for p in region.params} 

117 for n in walk_region(region): 

118 if id(n) in seen or n.op in ("core.yield", "core.tuple", "core.const"): 

119 continue 

120 seen.add(id(n)) 

121 if not n.op.startswith("tl."): 

122 raise KeyError(f"no ops_count rule for op {n.op!r}") 

123 base, p = n.op[3:], dict(n.attrs) 

124 c: Counter = Counter() 

125 if base == "pointwise": 

126 c = _scale(_marker_ops(p["f"]), _numel(n.type.layout)) 

127 elif base in ("reduce", "scan"): 

128 f = p["f"] 

129 nin = _numel(n.args[0].type.layout) 

130 if base == "reduce": 

131 lines = _numel(n.type.layout) 

132 else: 

133 size = n.args[0].type.layout.dim(p["dim"]).size 

134 lines = nin // size if size else 0 

135 folds = max(nin - lines, 0) 

136 if f in _RED_COMBINE: 

137 c[_RED_COMBINE[f]] = folds 

138 if f == "mean": 

139 c["div"] += lines if base == "reduce" else nin 

140 else: 

141 r = REDUCERS[f] 

142 lift = sum((_tree_ops(x) for x in r.lift), Counter()) 

143 combine = sum((_tree_ops(x) for x in r.combine), Counter()) 

144 project = _tree_ops(r.project) 

145 c = _scale(lift, nin) + _scale(combine, folds) 

146 c += _scale(project, lines if base == "reduce" else nin) 

147 elif base == "materialize": 

148 c["copy"] = _numel(n.type.layout) 

149 elif base == "take": 

150 c["take"] = _numel(n.type.layout) 

151 elif base == "scatter_add": 

152 nv = _numel(n.args[0].type.layout) 

153 c["scatter"] = nv 

154 c["add"] = nv 

155 elif base in ("argtopk", "argsort"): 

156 c[base] = _numel(n.args[0].type.layout) 

157 elif base == "fold": 

158 # per-step cost x step count, recursively (nested folds compose) 

159 state, element = tuple(p["state"]), tuple(p["element"]) 

160 if element: 

161 d0 = next(dd for dd in n.args[len(state)].type.dims if dd.name == p["dim"]) 

162 start, stop = d0.start, d0.stop 

163 else: 

164 start, stop = tuple(p["extent"]) 

165 sub = _region_ops_count(n.regions[0], fuse_mac, region_names(n.regions[0], state + element)) 

166 c = Counter({name: v * max(stop - start, 0) for name, v in sub.total.items()}) 

167 elif base in _FREE_OPS or base in LAYOUT_OP_NAMES: 

168 pass # closed forms, metadata, views, and broadcast declarations cost nothing 

169 else: 

170 raise KeyError(f"no ops_count rule for op {n.op!r}") 

171 per[names[id(n)]] = c 

172 

173 if fuse_mac: 

174 consumers: Counter = Counter() 

175 for n in walk_region(region): 

176 # outputs are not consumption: yields and the yielded tuple are 

177 # plumbing; a yielded tl node still consumes its own operands 

178 if n.op == "core.yield" or (n.op == "core.tuple" and n is yielded): 

179 continue 

180 for a in n.args: 

181 consumers[id(a)] += 1 

182 for n in walk_region(region): 

183 if n.op != "tl.reduce" or dict(n.attrs).get("f") != "sum" or len(n.args) != 1: 

184 continue 

185 src = n.args[0] 

186 if src.op != "tl.pointwise" or dict(src.attrs).get("f") != "mul": 

187 continue 

188 if consumers[id(src)] != 1: 

189 continue # the product is observable elsewhere; no fusion 

190 muls = per[names[id(src)]].pop("mul", 0) 

191 if muls: 

192 per[names[id(src)]]["mac"] = muls 

193 per[names[id(n)]].pop("add", None) 

194 

195 total: Counter = Counter() 

196 for c in per.values(): 

197 total += c 

198 return ProgramOps(per, +total)
