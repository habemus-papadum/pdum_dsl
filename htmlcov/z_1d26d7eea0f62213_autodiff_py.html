1"""Reverse-mode AD as a program transformation on the linear SSA IR. 

2 

3`grad(prog, target, input_layouts, seed=None)` returns (joint_program, 

4grads): the forward program truncated at `target`, extended with backward 

5instructions computing the cotangent of every forward variable, and a map 

6var -> gradient-var (None where no gradient flows). One backward pass yields 

7d(target)/d(v) for ALL v — reverse mode's native output. The generated 

8program is ordinary IR: run it with `ir.run`, inspect it, differentiate it 

9again. 

10 

11Contracts and conventions: 

12- Seed: scalar targets seed with 1 automatically; non-scalar targets REQUIRE 

13 `seed` (the name of an extra runtime input aligned with the target) — 

14 silent ones-seeding is a footgun we refuse. 

15- The tape is the program: adjoints reference forward variables directly 

16 (e.g. d(exp) reuses the forward output). 

17- Gradients carry their primal's COORDINATE charts and labels: AD 

18 differentiates value-with-respect-to-value, never with respect to 

19 coordinates, so the gradient entry for the sample at x = 0.75 um is 

20 labeled x = 0.75 um. This holds BY CONSTRUCTION: every contribution is 

21 restamped with its primal's charts/labels at the moment it is recorded 

22 (which also absorbs select's axis-compensation and promotion). Composite 

23 rules (decimate, diagonal) work on the bare lattice internally and are 

24 restamped on the way out. 

25- VALUE units transform as unit(dL/dv) = unit(L)/unit(v). Pass 

26 `target_unit` (the unit of the effective scalar) and gradients of inputs 

27 with declared value_units are annotated accordingly; a runtime seed 

28 should itself carry unit(L)/unit(target). 

29- Differentiability is structural: markers declare their partials 

30 (comparisons and iota declare none — the carrier discipline, enforced by 

31 the rule table rather than a type system for now). Cotangents reaching a 

32 gradient-free op are dropped. 

33- Fan-out accumulates: a var consumed n times receives n contributions, 

34 summed pointwise. 

35- Adjoints of layout ops are layout(+compute) programs: repeat†=reduce-sum, slice†=pad-0, pad†=slice (fill cotangent 

36 discarded), shift/flip/rename/split/merge are relabelings, 

37 select†=repeat-at-point+pad, window/stencil†=per-tap overlap-add, 

38 decimate†=zero-stuffing (materialize+merge; factor-divisible source 

39 domain only), 

40 diagonal†=masked embedding (binary only), scan(sum)†=reverse scan. 

41- Composite reducers differentiate via BPTT-as-IR: the state cotangent of a 

42 structured-state scan is itself a linear recurrence in reversed time, run 

43 as a generated matrix-linrec scan over derived Jacobian trees 

44 (composite_scan_adjoint); reduce† = embed-at-last, then scan†. 

45- The at-kink law (S.2, re-pinned at P4): the derivative table is one-sided 

46 and PARTITIONS — at a tie exactly one operand receives the cotangent, 

47 first-wins. Pointwise maximum/minimum select via the ge/gt asymmetry; 

48 reduce(max/min) derives as a chain of single-dim reduces with a 

49 first-occurrence mask, inheriting the partition. Gradient mass is 

50 conserved regardless of tie structure (test_at_kink pins the points). 

51- Primitive linearizations come from THE one derivative table 

52 (derivative.TABLE): a primitive's slope is registered once as a derived 

53 slope marker (f.d{i}) and applied over the operands — the same interface 

54 composite partials use. None = gradient-free.

74@lru_cache(maxsize=None) 

75def _revolve_cost(s: int, length: int) -> float: 

76 """Minimum recomputed forward-steps to reverse `length` steps with `s` 

77 checkpoint slots, under the revolve recursion (split at c: advance the 

78 head, reverse the tail with s-1 slots, reverse the head with s slots). 

79 A range is a store-all leaf — zero re-advance — once it fits in the 

80 available slots (length <= 1, or s >= length).""" 

81 if length <= 1 or s >= length: 

82 return 0.0 

83 if s < 1: 

84 return float("inf") # cannot subdivide a multi-step range with no slot 

85 return min(m + _revolve_cost(s - 1, length - m) + _revolve_cost(s, m) for m in range(1, length))

88@lru_cache(maxsize=None) 

89def _revolve_split(s: int, length: int) -> int: 

90 """Head length (steps to advance before the next checkpoint) for the 

91 optimal revolve split of `length` steps with `s` slots. This is exactly 

92 the Griewank & Walther binomial rule — with s slots and r recomputations 

93 one reverses up to C(s+r, s) steps — computed here by a memoized DP over 

94 the recompute-cost recurrence rather than the closed form, since fold 

95 lengths are modest at trace time. Ties are broken toward the LONGER head 

96 (advance as far as possible), matching revolve's convention. Only called 

97 for a genuine split: 1 <= s and 2 <= length and s < length.""" 

98 best = _revolve_cost(s, length) 

99 return max(m for m in range(1, length) if m + _revolve_cost(s - 1, length - m) + _revolve_cost(s, m) == best)

286 A = nd.args

526 sn: emit("fold", ops_, {**base_at, **extra_p, "out": ("final", i)}, regions=(step,)) 

527 for i, sn in enumerate(state_names) 

528 } 

529 

530 def leaf_backward(lo, hi, s0, cur, full): 

531 ops_, extra_p, se = seg_ops(tuple(s0[sn] for sn in state_names), lo, hi) 

532 sprev = {} 

533 for i, sn in enumerate(state_names): 

534 traj = emit("fold", ops_, {**base_at, **extra_p, "out": ("emit",)}, regions=(traj_steps[i],)) 

535 sh = emit("shift", (traj,), {"deltas": {dim: 1}}) 

536 sl = emit("slice", (sh,), {"ranges": {dim: (lo + 1, hi)}}) 

537 pd = emit("pad", (sl,), {"fill": 0.0, "extents": {dim: (lo, hi)}}) 

538 ri = emit("repeat", (s0[sn],), {"name": dim, "extent": (lo, hi)}) 

539 it = emit("iota", (pd,), {"name": dim}) 

540 sdims = extents_of(slayouts[sn]) + ((dim, (lo, hi)),) 

541 cs = emit("const", (), {"value": lo, "dims": sdims, "dtype": "int64"}) 

542 cs = restamp(cs, slayouts[sn]) # partial stamp: scan dim stays bare 

543 mask = emit("pointwise", (it, cs), {"f": "eq"}) 

544 sprev[sn] = emit("pointwise", (mask, ri, pd), {"f": "where"}) 

545 ybj = yb if full else emit("slice", (yb,), {"ranges": {dim: (lo, hi)}}) 

546 adj_ops = ( 

547 tuple(cur[sn] for sn in state_names) 

548 + tuple(flp(sprev[sn]) for sn in state_names) 

549 + tuple(flp(x) for x in se) 

550 + (flp(ybj),) 

551 ) 

552 for en, ev_ in zip(elem_names, elems): 

553 gn = gnode(en) 

554 if gn is None: 

555 continue 

556 fv = flp(emit("fold", adj_ops, {**adj_at, "out": ("emit",)}, regions=(adj_step(gn),))) 

557 if not full: 

558 fv = emit("pad", (fv,), {"fill": 0.0, "extents": {dim: (start, stop)}}) 

559 contribute(ev_, fv) 

560 return { 

561 sn: emit("fold", adj_ops, {**adj_at, "out": ("final", i)}, regions=(adj_base,)) 

562 for i, sn in enumerate(state_names) 

563 } 

564 

565 T = stop - start 

566 if fold_slots is not None: 

567 S = int(fold_slots) 

568 cur = zero_state() 

569 

570 def revolve(lo, hi, s, boundary): 

571 nonlocal cur 

572 span = hi - lo 

573 if span <= 1 or s >= span: 

574 cur = leaf_backward(lo, hi, boundary, cur, full=(lo == start and hi == stop)) 

575 return 

576 c_ = lo + _revolve_split(s, span) 

577 state_c = advance(boundary, lo, c_) 

578 revolve(c_, hi, s - 1, state_c) 

579 revolve(lo, c_, s, boundary) 

580 

581 revolve(start, stop, S, dict(zip(state_names, inits))) 

582 for sn, iv in zip(state_names, inits): 

583 if gnode(sn) is None: 

584 continue 

585 contribute(iv, cur[sn]) 

586 return 

587 

588 K = 1 if fold_segments is None else min(int(fold_segments), T) 

589 if K < 1: 

590 raise ValueError("fold_segments must be >= 1") 

591 if T % K: 

592 raise ValueError(f"fold_segments={K} must divide the fold extent {T} (pad the dim or pick a divisor)") 

593 L = T // K 

594 seg_start = [dict(zip(state_names, inits))] 

595 for j in range(K - 1): 

596 lo, hi = start + j * L, start + (j + 1) * L 

597 seg_start.append(advance(seg_start[-1], lo, hi)) 

598 cur = zero_state() 

599 for j in reversed(range(K)): 

600 lo, hi = start + j * L, start + (j + 1) * L 

601 cur = leaf_backward(lo, hi, seg_start[j], cur, full=(K == 1)) 

602 for sn, iv in zip(state_names, inits): 

603 if gnode(sn) is None: 

604 continue 

605 contribute(iv, cur[sn]) 

606 

607 def reduce_rule(nd, c): 

608 p = _thaw_params(dict(nd.attrs)) 

609 f = p["f"] 

610 dims = p["dims"] 

611 dim_names = (dims,) if isinstance(dims, str) else tuple(dims) 

612 if f not in RED and f in REDUCERS: 

613 (dim,) = dim_names 

614 ddim = lay(nd.args[0]).dim(dim) 

615 if ddim.size == 0: 

616 for e in nd.args: 

617 contribute(e, zeros_like(lay(e))) 

618 return 

619 lc = emit("strip_charts", (c,), {}, hint="lat") 

620 r = emit("repeat", (lc,), {"name": dim, "extent": (ddim.stop - 1, ddim.stop)}) 

621 yb = emit("pad", (r,), {"fill": 0.0, "extents": {dim: (ddim.start, ddim.stop)}}) 

622 composite_scan_adjoint(f, dim, nd.args, yb) 

623 return 

624 A = nd.args[0] 

625 a_shape = lay(A) 

626 if f == "sum": 

627 contribute(A, repeats_over(c, dim_names, a_shape)) 

628 elif f == "mean": 

629 r = repeats_over(c, dim_names, a_shape) 

630 count = 1 

631 for name in dim_names: 

632 count *= a_shape.dim(name).size 

633 nb = const_like(a_shape, float(count)) 

634 contribute(A, emit("pointwise", (r, nb), {"f": "div"})) 

635 elif f in ("max", "min"): 

636 plain = tuple(replace(d, chart=None, labels=None) for d in a_shape.dims) 

637 

638 def rep(v, d0): 

639 r = emit("repeat", (v,), {"name": d0.name, "extent": (d0.start, d0.stop)}, hint="km") 

640 if d0.level is not None: 

641 r = emit("bind", (r,), {"levels": {d0.name: d0.level}}, hint="km") 

642 return r 

643 

644 links = [] 

645 srcp, sdims = emit("strip_charts", (A,), {}, hint="kk"), plain 

646 for i, name in enumerate(dim_names): 

647 last = i == len(dim_names) - 1 

648 dst = nd if last else emit("reduce", (srcp,), {"f": f, "dims": (name,)}, hint="kr") 

649 dstp = emit("strip_charts", (dst,), {}, hint="kk") if last else dst 

650 links.append((srcp, sdims, name)) 

651 sdims = tuple(d for d in sdims if d.name != name) 

652 srcp = dstp 

653 prev, cotan = srcp, emit("strip_charts", (c,), {}, hint="kc") 

654 for srcp, sdims, name in reversed(links): 

655 d0 = next(d for d in sdims if d.name == name) 

656 lay_ = _dense_like(sdims) 

657 one, zero = const_like(lay_, 1.0), const_like(lay_, 0.0) 

658 m = emit("pointwise", (srcp, rep(prev, d0)), {"f": "eq"}, hint="kt") 

659 mnum = emit("pointwise", (m, one, zero), {"f": "where"}, hint="kn") 

660 s = emit("scan", (mnum,), {"f": "sum", "dim": name}, hint="ks") 

661 fw = emit("pointwise", (s, one), {"f": "eq"}, hint="kf") 

662 first = emit("pointwise", (m, fw), {"f": "mul"}, hint="kw") 

663 cotan = emit("pointwise", (rep(cotan, d0), first), {"f": "mul"}, hint="kg") 

664 prev = srcp 

665 contribute(A, cotan) 

666 else: 

667 raise NotImplementedError(f"reduce({f}) has no adjoint rule yet") 

668 

669 def scan_rule(nd, c): 

670 p = dict(nd.attrs) 

671 if p["f"] in REDUCERS and p["f"] not in RED: 

672 composite_scan_adjoint(p["f"], p["dim"], nd.args, c) 

673 return 

674 if p["f"] != "sum": 

675 raise NotImplementedError("only scan(sum) is differentiable so far") 

676 dim = p["dim"] 

677 f1 = emit("flip", (c,), {"name": dim}) 

678 s1 = emit("scan", (f1,), {"f": "sum", "dim": dim}) 

679 contribute(nd.args[0], emit("flip", (s1,), {"name": dim}))

681 def tap_rule(nd, c, p): 

682 name, k_name = p["name"], p.get("k_name") or f"{p['name']}_k" 

683 dilation = p.get("dilation", 1) 

684 A = nd.args[0] 

685 src = lay(A).dim(name) 

686 kdim = lay(nd).dim(k_name) 

687 for kappa in range(kdim.start, kdim.stop): 

688 t1 = emit("select", (c,), {"coords": {k_name: kappa}}) 

689 t2 = emit("shift", (t1,), {"deltas": {name: kappa * dilation}}) 

690 anchor = lay(nd).dim(name) 

691 lo = max(src.start, anchor.start + kappa * dilation) 

692 hi = min(src.stop, anchor.stop + kappa * dilation) 

693 if hi <= lo: 

694 lo = hi = src.start 

695 t3 = emit("slice", (t2,), {"ranges": {name: (lo, hi)}}) 

696 t4 = emit("pad", (t3,), {"fill": 0.0, "extents": {name: (src.start, src.stop)}}) 

697 contribute(A, t4) 

698 

699 def decimate_rule(nd, c, p): 

700 name, f = p["name"], p["factor"] 

701 A = nd.args[0] 

702 c = emit("strip_charts", (c,), {}, hint="lat") 

703 src = lay(A).dim(name) 

704 phase = src.delta_to_lattice(p.get("phase", 0)) % f 

705 s, e = src.start, src.stop 

706 outd = lay(nd).dim(name) 

707 if (e - s) % f or (e - s) // f != outd.size: 

708 raise NotImplementedError( 

709 f"decimate adjoint needs a factor-divisible domain: [{s}, {e}) with factor {f} (pad the source first)" 

710 ) 

711 slot = (phase - s) % f 

712 used = {d.name for d in lay(nd).dims} 

713 

714 def fresh_dim(base): 

715 i = 0 

716 while f"_{base}{i}" in used: 

717 i += 1 

718 used.add(f"_{base}{i}") 

719 return f"_{base}{i}" 

720 

721 cname = fresh_dim("c") 

722 ph = fresh_dim("ph") 

723 r0 = emit("rename", (c,), {"mapping": {name: cname}}) 

724 r1 = emit("repeat", (r0,), {"name": ph, "extent": (0, f)}) 

725 i1 = emit("iota", (r1,), {"name": ph}) 

726 r1_dims = tuple((cname if d.name == name else d.name, (d.start, d.stop)) for d in lay(nd).dims) + ( 

727 (ph, (0, f)), 

728 ) 

729 cp = emit("const", (), {"value": slot, "dims": r1_dims, "dtype": "int64"}) 

730 m = emit("pointwise", (i1, cp), {"f": "eq"}) 

731 z0 = emit("const", (), {"value": 0.0, "dims": r1_dims}) 

732 w = emit("pointwise", (m, r1, z0), {"f": "where"}) 

733 others = tuple(nm for nm, _ in r1_dims if nm not in (cname, ph)) 

734 mo = emit("materialize", (w,), {"order": others + (cname, ph)}) 

735 mg = emit("merge", (mo,), {"parts": (cname, ph), "name": name, "start": s}) 

736 contribute(A, mg) 

737 

738 def diagonal_rule(nd, c, p): 

739 parts = tuple(p["parts"]) 

740 if len(parts) != 2: 

741 raise NotImplementedError("n-ary diagonal adjoint not written yet") 

742 x, y = parts 

743 z = p["name"] 

744 A = nd.args[0] 

745 xdom, ydom = lay(A).dim(x), lay(A).dim(y) 

746 if lay(nd).dim(z).size == 0: 

747 contribute(A, zeros_like(lay(A))) 

748 return 

749 c = emit("strip_charts", (c,), {}, hint="lat") 

750 r1 = c if z == x else emit("rename", (c,), {"mapping": {z: x}}) 

751 r2 = emit("repeat", (r1,), {"name": y, "extent": (ydom.start, ydom.stop)}) 

752 ix = emit("iota", (r2,), {"name": x}) 

753 iy = emit("iota", (r2,), {"name": y}) 

754 m = emit("pointwise", (ix, iy), {"f": "eq"}) 

755 r2_dims = tuple((x if d.name == z else d.name, (d.start, d.stop)) for d in lay(nd).dims) + ( 

756 (y, (ydom.start, ydom.stop)), 

757 ) 

758 z0 = emit("const", (), {"value": 0.0, "dims": r2_dims}) 

759 w = emit("pointwise", (m, r2, z0), {"f": "where"}) 

760 zdim = lay(nd).dim(z) 

761 if (zdim.start, zdim.stop) != (xdom.start, xdom.stop): 

762 w = emit("pad", (w,), {"fill": 0.0, "extents": {x: (xdom.start, xdom.stop)}}) 

763 contribute(A, w)

765 def layout_rule(nd, c): 

766 base = nd.op[3:] 

767 A = nd.args[0] 

768 a_shape = lay(A) 

769 p = _thaw_params(dict(nd.attrs)) 

770 if base == "slice": 

771 extents = {nm: (a_shape.dim(nm).start, a_shape.dim(nm).stop) for nm in p["ranges"]} 

772 contribute(A, emit("pad", (c,), {"fill": 0.0, "extents": extents})) 

773 elif base == "pad": 

774 ranges = {nm: (a_shape.dim(nm).start, a_shape.dim(nm).stop) for nm in p["extents"]} 

775 contribute(A, emit("slice", (c,), {"ranges": ranges})) 

776 elif base == "shift": 

777 deltas = {nm: -a_shape.dim(nm).delta_to_lattice(v) for nm, v in p["deltas"].items()} 

778 contribute(A, emit("shift", (c,), {"deltas": deltas})) 

779 elif base == "flip": 

780 contribute(A, emit("flip", (c,), {"name": p["name"]})) 

781 elif base == "rename": 

782 inv = {new: old for old, new in p["mapping"].items()} 

783 contribute(A, emit("rename", (c,), {"mapping": inv})) 

784 elif base == "repeat": 

785 contribute(A, emit("reduce", (c,), {"f": "sum", "dims": (p["name"],)})) 

786 elif base == "select": 

787 cur = c 

788 coord_names = list(p["coords"]) 

789 for nm, coord in p["coords"].items(): 

790 i = a_shape.dim(nm).to_lattice(coord) 

791 cur = emit("repeat", (cur,), {"name": nm, "extent": (i, i + 1)}) 

792 extents = {nm: (a_shape.dim(nm).start, a_shape.dim(nm).stop) for nm in coord_names} 

793 contribute(A, emit("pad", (cur,), {"fill": 0.0, "extents": extents})) 

794 elif base == "split": 

795 parts = tuple(p["parts"]) 

796 others = tuple(d.name for d in lay(nd).dims if d.name not in parts) 

797 mo = emit("materialize", (c,), {"order": others + parts}) 

798 contribute( 

799 A, 

800 emit("merge", (mo,), {"parts": parts, "name": p["name"], "start": a_shape.dim(p["name"]).start}), 

801 ) 

802 elif base == "merge": 

803 # ORDERED parts: pre-tupled so the canonical attr sort cannot 

804 # reorder the mixed-radix nesting (the lifting packer precedent) 

805 parts = tuple((nm, (a_shape.dim(nm).start, a_shape.dim(nm).stop)) for nm in p["parts"]) 

806 contribute(A, emit("split", (c,), {"name": p["name"], "parts": parts})) 

807 elif base in ("window", "stencil"): 

808 tap_rule(nd, c, p) 

809 elif base == "decimate": 

810 decimate_rule(nd, c, p) 

811 elif base == "diagonal": 

812 diagonal_rule(nd, c, p) 

813 elif base in ( 

814 "strip_charts", 

815 "with_charts", 

816 "with_labels", 

817 "with_value_units", 

818 "bind", 

819 "simplify", 

820 "materialize", 

821 ): 

822 contribute(A, c) # value-preserving metadata / identity copy 

823 else: 

824 raise NotImplementedError(f"no adjoint rule for {base!r}") 

825 

826 # ---- reverse walk -------------------------------------------------- 

827 for nd in reversed(fwd): 

828 c = finalize(nd) 

829 if c is None: 

830 continue 

831 if nd.op in ("core.param", "core.const") or not nd.op.startswith("tl."): 

832 continue 

833 base = nd.op[3:] 

834 if base in ("const", "iota", "random"): 

835 continue 

836 if base in ("argtopk", "argsort"): 

837 continue 

838 if base == "repeat_like": 

839 have = {d.name for d in lay(nd.args[0]).dims} 

840 added = tuple(d.name for d in lay(nd).dims if d.name not in have) 

841 gv = emit("reduce", (c,), {"f": "sum", "dims": added}) if added else c 

842 contribute(nd.args[0], gv) 

843 elif base == "round_to": 

844 if dict(nd.attrs).get("grad", "straight_through") != "zero": 

845 contribute(nd.args[0], c) 

846 elif base == "take": 

847 d = lay(nd.args[0]).dim(dict(nd.attrs)["dim"]) 

848 survivors = {x.name for x in lay(nd.args[0]).dims if x.name != d.name} 

849 spliced = tuple(nm for nm in lay(nd.args[1]).names if nm not in survivors) 

850 gv = emit( 

851 "scatter_add", 

852 (c, nd.args[1]), 

853 {"dim": d.name, "extent": (d.start, d.stop), "over": spliced}, 

854 ) 

855 contribute(nd.args[0], gv) 

856 elif base == "scatter_add": 

857 gv = emit("take", (c, nd.args[1]), {"dim": dict(nd.attrs)["dim"]}) 

858 contribute(nd.args[0], gv) 

859 elif base == "pointwise": 

860 pw_rule(nd, c) 

861 elif base == "reduce": 

862 reduce_rule(nd, c) 

863 elif base == "scan": 

864 scan_rule(nd, c) 

865 elif base == "fold": 

866 fold_rule(nd, c) 

867 else: 

868 layout_rule(nd, c) 

869 

870 # ---- outputs ------------------------------------------------------- 

871 grads: dict = {} 

872 req = tuple(wrt) if wrt is not None else tuple(names[id(p_)] for p_ in region.params) 

873 ynodes_out = [tnode] 

874 outputs = [target] 

875 for v in req: 

876 node_v = by_name.get(v) 

877 if node_v is None: 

878 raise KeyError(f"wrt name {v!r} is not defined by the program") 

879 gn = finalize(node_v) 

880 if gn is not None and target_unit is not None: 

881 uv = sigs[v].unit if v in sigs else None 

882 gu = target_unit if uv is None or uv == ONE else target_unit / uv 

883 gn = emit("with_value_units", (gn,), {"value_units": gu}, hint="vu") 

884 if gn is None: 

885 grads[v] = None 

886 continue 

887 grads[v] = jnames[id(gn)] 

888 ynodes_out.append(gn) 

889 outputs.append(jnames[id(gn)]) 

890 y = ynodes_out[0] if len(ynodes_out) == 1 else b.emit("core.tuple", *ynodes_out) 

891 joint_params = tuple(region.params) + ((seed_param,) if seed_param is not None else ()) 

892 joint = Region(params=joint_params, body=(b.emit("core.yield", y),)) 

893 return RegionGrad(joint, grads, jnames, tuple(outputs), {jnames[id(x)]: x for x in ynodes_out})

896def numeric_grad(region, target: str, wrt_var: str, inputs: dict, names, eps: float = 1e-6): 

897 """Central finite differences of a SCALAR target w.r.t. one input tensor, 

898 on the region face. Rebuilds perturbed inputs via from_numpy with the 

899 base tensor's stamps — use simple inputs in FD tests.""" 

900 import numpy as np 

901 

902 from .dialect import run_named 

903 from .tensor import Tensor 

904 

905 base = inputs[wrt_var] 

906 arr = base.to_numpy().astype(np.float64) 

907 tnames = base.names 

908 charts = {d.name: d.chart for d in base.layout.dims if d.chart is not None} 

909 labels = {d.name: d.labels for d in base.layout.dims if d.labels is not None} 

910 levels = {d.name: d.level for d in base.layout.dims if d.level is not None} 

911 

912 def rebuild(pert): 

913 t = Tensor.from_numpy(pert, tnames) 

914 if charts: 

915 t = t.with_charts(**charts) 

916 if labels: 

917 t = t.with_labels(**labels) 

918 if levels: 

919 t = t.bind(**levels) 

920 if base.value_units is not None: 

921 t = t.with_value_units(base.value_units) 

922 return t 

923 

924 g = np.zeros_like(arr) 

925 for idx in np.ndindex(*arr.shape): 

926 out = [] 

927 for sign in (+1, -1): 

928 pert = arr.copy() 

929 pert[idx] += sign * eps 

930 vals = run_named(region, {**inputs, wrt_var: rebuild(pert)}, names) 

931 out.append(float(vals[target].to_numpy())) 

932 g[idx] = (out[0] - out[1]) / (2 * eps) 

933 return g
