1"""Buffer: rank-1, featureless storage — plus the read seam. 

2 

3A Buffer is a pointer-like handle plus an address space and a length in 

4bytes. It has no shape, no strides, no dtype — all of that lives in the 

5Layout/Tensor layer. DLPack would be the interchange format for real device 

6buffers; host buffers carry a memoryview so the layout math can be 

7*exercised* in tests; device buffers are just described (data=None). 

8 

9`read(loc, dtype)` is the single seam through which values leave a buffer. 

10A tensor is the composition value(coords) = buffer.read(loc(coords)) — the 

11layout owns the affine part, the buffer owns the dereference. That 

12factorization is what makes FunctionalBuffer possible: 

13 

14`FunctionalBuffer` owns NO memory and declares its read as an exact affine 

15function of the byte location: read(loc) = const + coeff * (loc // scale), 

16with rational coefficients, cast to the requesting dtype only at the read. 

17It is the tight form of `iota` (the identity table, declared instead of 

18materialized) — and the closure invariant comes free: every view op rewrites 

19only the layout, never the buffer, and affine ∘ affine = affine, so 

20iota-ness cannot be destroyed by any layout operation. A compiler recognizes 

21the closed form by the buffer's type. 

22""" 

23 

24from __future__ import annotations 

25 

26from dataclasses import dataclass 

27from fractions import Fraction 

28 

29import numpy as np 

30 

31 

32@dataclass(frozen=True, eq=False) 

33class Buffer: 

34 nbytes: int 

35 device: str = "cpu" 

36 data: memoryview | None = None # host bytes when accessible, else None 

37 

38 def __post_init__(self) -> None: 

39 if self.nbytes < 0: 

40 raise ValueError("nbytes must be non-negative") 

41 if self.data is not None and len(self.data) != self.nbytes: 

42 raise ValueError(f"data length {len(self.data)} != nbytes {self.nbytes}") 

43 

44 @classmethod 

45 def from_bytes(cls, raw, device: str = "cpu") -> "Buffer": 

46 mv = memoryview(raw).cast("B") 

47 return cls(nbytes=len(mv), device=device, data=mv) 

48 

49 @classmethod 

50 def allocate(cls, nbytes: int, device: str = "cpu") -> "Buffer": 

51 return cls.from_bytes(bytearray(nbytes), device=device) 

52 

53 @property 

54 def is_host(self) -> bool: 

55 return self.data is not None 

56 

57 def read(self, loc: int, dtype) -> object: 

58 """The read seam: one value of `dtype` at byte `loc`.""" 

59 if self.data is None: 

60 raise RuntimeError(f"buffer on {self.device} is not host-readable") 

61 return np.frombuffer(self.data, dtype, count=1, offset=loc)[0] 

62 

63 def write(self, loc: int, dtype, value) -> None: 

64 """The write seam: the read seam's mirror — one value of `dtype` 

65 into byte `loc`.""" 

66 if self.data is None: 

67 raise RuntimeError(f"buffer on {self.device} is not host-writable") 

68 if self.data.readonly: 

69 raise RuntimeError("buffer bytes are read-only") 

70 np.frombuffer(self.data, dtype, count=1, offset=loc)[0] = value 

71 

72 def __repr__(self) -> str: 

73 return f"Buffer({self.nbytes}B @ {self.device})" 

74 

75 

76def host_view(carr) -> memoryview: 

77 """Best-effort zero-copy byte view of a C-contiguous array; falls back 

78 to a copy when the buffer protocol refuses the cast (e.g. some 

79 structured dtypes). The single home of this fallback.""" 

80 try: 

81 return memoryview(carr).cast("B") 

82 except ValueError, TypeError: 

83 return memoryview(bytearray(carr.tobytes())) 

84 

85 

86@dataclass(frozen=True, eq=False) 

87class FunctionalBuffer(Buffer): 

88 """No memory: read(loc) = const + coeff * (loc // scale), exactly. 

89 

90 Coefficients are exact rationals; the cast to the requesting dtype 

91 happens only at the read — representation never enters the semantics. 

92 `nbytes` is the extent the buffer *would* occupy if materialized, so 

93 footprint/check semantics are identical to the tabulated form.""" 

94 

95 scale: int = 1 # bytes per lattice step (the itemsize of the virtual table) 

96 coeff: Fraction = Fraction(1) 

97 const: Fraction = Fraction(0) 

98 

99 def __post_init__(self) -> None: 

100 super().__post_init__() 

101 if self.scale <= 0: 

102 raise ValueError("scale must be a positive byte count") 

103 object.__setattr__(self, "coeff", Fraction(self.coeff)) 

104 object.__setattr__(self, "const", Fraction(self.const)) 

105 

106 def write(self, loc: int, dtype, value) -> None: 

107 raise RuntimeError("a functional buffer owns no memory — nothing to write") 

108 

109 def read(self, loc: int, dtype) -> object: 

110 q, r = divmod(loc, self.scale) 

111 if r: 

112 raise ValueError(f"misaligned functional read at byte {loc} (scale {self.scale})") 

113 value = self.const + self.coeff * q 

114 dt = np.dtype(dtype) 

115 if dt.kind in "iub": 

116 if value.denominator != 1: 

117 raise ValueError(f"non-integer value {value} read as {dt}") 

118 return dt.type(int(value)) 

119 return dt.type(float(value)) 

120 

121 def __repr__(self) -> str: 

122 return f"FunctionalBuffer({self.nbytes}B, read = {self.const} + {self.coeff}*(loc//{self.scale}))"
