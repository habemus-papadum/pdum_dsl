1"""The reference evaluator: the value tier's oracle (design 200 §1.1). 

2 

3Renders a legalized Region to *readable* Python source and executes it with 

4the exact ABI every real target will use: ``kernel(staging, leaves)``, every 

5input — captures AND scalar arguments — arriving as bytes in the staging 

6buffer at the offsets the plan chose. Deliberately small, deliberately slow: 

7this is the twin every future device backend is differential-tested 

8against, never a production path. 

9 

10**Oracle execution is always spelled**: ``reference(f)(...)``. A plain call 

11on a kind with no routed backend refuses (registry.backend_for); it does not 

12degrade to interpretation. 

13 

14Reference semantics, stated once: 

15 

16- ``core.if`` renders as a real ``if``/``else`` — branches are LAZY, so the 

17 guard-then-divide idiom is safe; each node is emitted in the deepest 

18 region that dominates all its uses. 

19- Python has one float; ``f32`` computes as f64 here. Narrowing gets real in 

20 device backends. 

21- Numeric policy (distilled, amended by the numpy-authority ruling): floats 

22 compute on **numpy scalars** — IEEE NON-TRAPPING, like a device: 0/0 

23 flows as nan, sqrt(-1) is nan, never a Python exception. TRUNC integer 

24 div/mod via exact helpers (float rounding would lose precision past 

25 2^53); float mod is ``np.fmod`` (sign of the dividend). 

26- Composite (record-typed) *arguments* are refused loudly: their leaf slots 

27 exist in the plan, but the IR keeps one logical ``core.param``. 

28"""

40_BIN = {"core.add": "+", "core.sub": "-", "core.mul": "*", "core.div": "/", "core.mod": "%", "core.pow": "**"} 

41_PREDS = {"lt": "<", "gt": ">", "le": "<=", "ge": ">=", "eq": "==", "ne": "!="} 

42_CASTS = {"f64": "float", "f32": "float", "i64": "int", "i32": "int", "u64": "int", "u32": "int", "bool": "bool"}

45def _unpack(fmt: str, offset) -> str: 

46 """Float slots arrive as np.float64 so the WHOLE float dataflow runs on 

47 numpy scalars — IEEE non-trapping (0/0 flows as nan, like a device; the 

48 210 amendment). Integer slots stay Python ints (exact, trunc-div policy 

49 unchanged).""" 

50 base = f"_u({fmt!r}, staging, {offset})[0]" 

51 return f"np.float64({base})" if fmt.endswith(("d", "f")) else base

54def render(region: Region, plan: PackPlan, backend=None, name: str = "kernel") -> str: 

55 """Legalized Region -> Python source. One assignment per node, DAG-shared 

56 nodes emitted once in their OWNER region (the dominator-placed walker in 

57 ``render`` — the lazy-branch rule), names dense in topo order.""" 

58 args_by_index = {s.source.index: s for s in plan.slots if s.source.root == "arg" and not s.source.sub} 

59 

60 def expr_of(node: Node, names: dict) -> str: 

61 attrs = dict(node.attrs) 

62 arg = [names[id(a)] for a in node.args] 

63 if node.op == "core.param": 

64 spec = args_by_index.get(attrs["index"]) 

65 if spec is None: 

66 raise VerifyError( 

67 f"argument {attrs['index']} has no scalar slot — composite (record-typed) " 

68 "arguments are not marshalable yet; pass fields as separate arguments" 

69 ) 

70 return _unpack(spec.dest.fmt, spec.dest.offset) 

71 if node.op == "abi.slot": 

72 return _unpack(attrs["fmt"], attrs["offset"]) 

73 if node.op == "core.const": 

74 v = attrs["value"] 

75 if isinstance(v, float) and not math.isfinite(v): 

76 return f"float({str(v)!r})" # repr(inf) is not a Python literal 

77 return repr(v) 

78 if node.op in _BIN: 

79 if node.op in ("core.div", "core.mod") and isinstance(node.type, Scalar) and node.type.kind[0] in "iu": 

80 return f"{'_tdiv' if node.op == 'core.div' else '_tmod'}({arg[0]}, {arg[1]})" 

81 if node.op == "core.mod": 

82 return f"np.fmod({arg[0]}, {arg[1]})" # sign of the dividend (numeric policy) 

83 return f"{arg[0]} {_BIN[node.op]} {arg[1]}" 

84 if node.op == "core.neg": 

85 return f"-{arg[0]}" 

86 if node.op == "core.cmp": 

87 return f"{arg[0]} {_PREDS[attrs['pred']]} {arg[1]}" 

88 if node.op == "core.cast": 

89 to = attrs["to"] 

90 if not isinstance(to, Scalar): 

91 raise VerifyError(f"the reference evaluator cannot cast to {to!r}") 

92 return f"{_CASTS[to.kind]}({arg[0]})" 

93 if node.op == "core.select": 

94 return f"({arg[1]} if {arg[0]} else {arg[2]})" 

95 if node.op in ("core.vec", "core.tuple"): 

96 return "(" + ", ".join(arg) + ("," if len(arg) == 1 else "") + ")" 

97 if node.op == "core.extract": 

98 return f"{arg[0]}[{attrs['index']}]" 

99 table = backend.code_for_op if backend is not None else CODE_FOR_OP 

100 template = table.get(node.op) # surface D: THIS record's spellings (extend()-local) 

101 if template: 

102 return template.format(*arg) 

103 if template is None and node.op in table: # spell(None) claims native support this renderer lacks 

104 raise VerifyError(f"{node.op!r} was spelled None ('native') but the reference has no native handling") 

105 raise VerifyError(f"the reference evaluator has no rendering for {node.op!r}") 

106 

107 def statement(node, nm): 

108 return f"{nm[id(node)]} = {expr_of(node, nm)}" 

109 

110 def branch_join(node, nm, result_of, emit_block, path, ind): 

111 res = nm[id(node)] 

112 out = [f"{ind}if {nm[id(node.args[0])]}:"] 

113 out += emit_block((*path, (id(node), 0)), ind + " ") 

114 out.append(f"{ind} {res} = {result_of(0)}") 

115 out.append(f"{ind}else:") 

116 out += emit_block((*path, (id(node), 1)), ind + " ") 

117 out.append(f"{ind} {res} = {result_of(1)}") 

118 return out 

119 

120 def loop_join(node, nm, result_of, emit_block, path, ind): 

121 res, (lo, hi, init) = nm[id(node)], (nm[id(a)] for a in node.args) 

122 iv, carry = node.regions[0].params 

123 out = [f"{ind}{res} = {init}"] # res IS the carry between iterations 

124 out.append(f"{ind}for {nm[id(iv)]} in range({lo}, {hi}):") 

125 out.append(f"{ind} {nm[id(carry)]} = {res}") 

126 out += emit_block((*path, (id(node), 0)), ind + " ") 

127 out.append(f"{ind} {res} = {result_of(0)}") 

128 if dict(node.attrs).get("exit"): # 300: (carry, done) — the host takes the break license 

129 out.append(f"{ind} {res}, _done = {res}") 

130 out.append(f"{ind} if _done: break") 

131 return out 

132 

133 lines, names, result = emit_dominated(region, statement, branch_join, indent=" ", loop=loop_join) 

134 body = "\n".join(lines) or " pass" 

135 head = ( 

136 "import math\n" # user-registered ops may spell math.* (surface D) 

137 "import numpy as np\nfrom struct import unpack_from as _u\n\n" 

138 "def _tdiv(a, b): # exact trunc division (numeric policy: C semantics)\n" 

139 " q = a // b\n" 

140 " return q + 1 if q < 0 and q * b != a else q\n\n" 

141 "def _tmod(a, b):\n" 

142 " return a - _tdiv(a, b) * b\n\n" 

143 f"def {name}(staging, leaves):\n" 

144 ) 

145 return f"{head}{body}\n return {names[id(result)]}\n"

148def compile_source(source: str, name: str = "kernel"): 

149 ns: dict = {} 

150 exec(compile(source, f"<pdum-reference:{name}>", "exec"), ns) # noqa: S102 — this IS the oracle 

151 artifact = ns[name] 

152 artifact.__pdum_source__ = source # the artifact carries its own listing 

153 return artifact

169def install(registry) -> None: 

170 """Register the oracle. NEVER default, NEVER routed: reference execution

172 OWN record copy with a fresh spelling table."""

178def reference(target, *, registry=None): 

179 """The spelled oracle door: ``reference(f)(args...)`` runs ``f`` on the 

180 reference evaluator through the same two-tier dispatch as any backend 

181 (specialization cache, guards, marshaling — the thesis machinery is 

182 exercised, only the artifact is the readable Python twin).""" 

183 if registry is None: 

184 from .registry import DEFAULT as registry # noqa: N811 

185 

186 def run(*args): 

187 return registry.dispatch(target, args, backend="reference") 

188 

189 return run
