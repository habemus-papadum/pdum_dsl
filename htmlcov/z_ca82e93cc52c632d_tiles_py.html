1"""The tile-tier flagships (320 §10.3) — hand-written tile programs beside 

2their naive twins. 

3 

4Each entry is one PER-TILE BODY (the binding law hands a kernel 

5pre-selected tiles; these regions are that body, so the grid is out of 

6frame) built op-by-op with the Builder — the honest authoring surface 

7until a ``@tile`` front-end exists. The naive twin is the erased idiom: 

8the same denotation with no splits, no stages, no tile-fold. Both run on 

9``run_region`` today — the tile tier's day-one oracle — and the 

10differential between them is gated by the DECLARED licenses (gemm's 

11k-sum re-bracketing; flash's online-softmax lemma), never by an ad-hoc 

12tolerance. The stencil has no licensed deviation, so its twin must 

13agree to the bit. 

14"""

32@dataclass(frozen=True) 

33class TileFlagship: 

34 region: Region # the tile-tier program (checked at the tile tier) 

35 naive: Region # the erased-idiom twin — the denotation 

36 inputs: dict[str, Tensor] # positional per param, both regions 

37 oracle: Callable # dict[str, np.ndarray] -> np.ndarray

40 return {k: v.to_numpy() for k, v in self.inputs.items()}

43def gemm_tile(MI: int = 8, NI: int = 6, K: int = 8, KI: int = 2, seed: int = 13) -> TileFlagship: 

44 """320 §2's flagship: acc carried through a fold over the k-tiles, both 

45 operand slices STAGED to "shared", the contraction spelled as 

46 repeat_like/mul/reduce. The accumulator is registers by ABSENCE — it is 

47 just the carry.""" 

48 KO = K // KI 

49 rng = np.random.default_rng(seed) 

50 A = Tensor.from_numpy(rng.standard_normal((MI, K)), ("mi", "k")) 

51 B = Tensor.from_numpy(rng.standard_normal((K, NI)), ("k", "ni")) 

52 

53 ops = {**CORE_OPS, **TL_OPS} 

54 b = Builder(ops) 

55 pa = b.param(0, tensor_type_of_layout(A.layout)) 

56 pb = b.param(1, tensor_type_of_layout(B.layout)) 

57 at = b.emit("tl.split", pa, name="k", parts=(("ko", KO), ("ki", KI))) 

58 bt = b.emit("tl.split", pb, name="k", parts=(("ko", KO), ("ki", KI))) 

59 acc0 = b.emit("tl.const", value=0.0, dims=(("mi", MI), ("ni", NI))) 

60 

61 # the step: (acc, a-slice, b-slice) at one ko -> next acc 

62 sb = Builder(ops) 

63 p_acc = sb.param(0, acc0.type) 

64 p_a = sb.param(1, _minus_dim(at.type, "ko")) 

65 p_b = sb.param(2, _minus_dim(bt.type, "ko")) 

66 a_s = sb.emit("tl.stage", p_a, level="shared") 

67 b_s = sb.emit("tl.stage", p_b, level="shared") 

68 prod = sb.emit("tl.pointwise", sb.emit("tl.repeat_like", a_s, b_s), sb.emit("tl.repeat_like", b_s, a_s), f="mul") 

69 part = sb.emit("tl.reduce", prod, dims=("ki",), f="sum") 

70 nxt = sb.emit("tl.pointwise", p_acc, part, f="add") 

71 step = Region(params=(p_acc, p_a, p_b), body=(sb.emit("core.yield", nxt),)) 

72 

73 fold = b.emit( 

74 "tl.fold", acc0, at, bt, regions=(step,), dim="ko", state=("acc",), element=("a", "b"), out=("final", 0) 

75 ) 

76 region = Region(params=(pa, pb), body=(b.emit("core.yield", fold),)) 

77 

78 # the naive twin: contract, unsplit and unstaged 

79 nb = Builder(ops) 

80 na = nb.param(0, tensor_type_of_layout(A.layout)) 

81 nnb = nb.param(1, tensor_type_of_layout(B.layout)) 

82 nprod = nb.emit("tl.pointwise", nb.emit("tl.repeat_like", na, nnb), nb.emit("tl.repeat_like", nnb, na), f="mul") 

83 nred = nb.emit("tl.reduce", nprod, dims=("k",), f="sum") 

84 naive = Region(params=(na, nnb), body=(nb.emit("core.yield", nred),)) 

85 

86 return TileFlagship( 

87 region=check_tier(region, "tile"), 

88 naive=naive, 

89 inputs={"a": A, "b": B}, 

90 oracle=lambda inp: inp["a"] @ inp["b"], 

91 )

94def stencil_tile(MI: int = 6, NI: int = 8, alpha: float = 0.1, seed: int = 17) -> TileFlagship: 

95 """The fused stencil chain (K-D) as one tile body: the halo tile stages 

96 to "shared" once, the 5-point Laplacian is four shifted slices over the 

97 STAGED copy, and the whole chain (nsum - 4u, u + alpha*lap) fuses as 

98 pointwise — one load, one tile of work. The twin runs the same chain on 

99 the unstaged halo. Domains: the halo is [0, MI+2) x [0, NI+2); the 

100 interior [1, MI+1) x [1, NI+1) is the tile's own patch.""" 

101 rng = np.random.default_rng(seed) 

102 U = Tensor.from_numpy(rng.standard_normal((MI + 2, NI + 2)), ("x", "y")) 

103 

104 ops = {**CORE_OPS, **TL_OPS} 

105 

106 def body(bld, halo): 

107 # neighbor(dx, dy): u[x+dx, y+dy] over the interior, via shift+slice 

108 def neighbor(dx, dy): 

109 sh = bld.emit("tl.shift", halo, deltas=(("x", -dx), ("y", -dy))) 

110 return bld.emit("tl.slice", sh, ranges=(("x", (1, MI + 1)), ("y", (1, NI + 1)))) 

111 

112 center = neighbor(0, 0) 

113 ew = bld.emit("tl.pointwise", neighbor(0, 1), neighbor(0, -1), f="add") 

114 ns = bld.emit("tl.pointwise", neighbor(1, 0), neighbor(-1, 0), f="add") 

115 nsum = bld.emit("tl.pointwise", ew, ns, f="add") 

116 four = bld.emit("core.const", type=f64, value=4.0) 

117 lap = bld.emit("tl.pointwise", nsum, bld.emit("tl.pointwise", four, center, f="mul"), f="sub") 

118 al = bld.emit("core.const", type=f64, value=alpha) 

119 return bld.emit("tl.pointwise", center, bld.emit("tl.pointwise", al, lap, f="mul"), f="add") 

120 

121 b = Builder(ops) 

122 ph = b.param(0, tensor_type_of_layout(U.layout)) 

123 staged = b.emit("tl.stage", ph, level="shared") 

124 region = Region(params=(ph,), body=(b.emit("core.yield", body(b, staged)),)) 

125 

126 nb = Builder(ops) 

127 nh = nb.param(0, tensor_type_of_layout(U.layout)) 

128 naive = Region(params=(nh,), body=(nb.emit("core.yield", body(nb, nh)),)) 

129 

130 def oracle(inp): 

131 u = inp["u"] 

132 lap = u[:-2, 1:-1] + u[2:, 1:-1] + u[1:-1, :-2] + u[1:-1, 2:] - 4 * u[1:-1, 1:-1] 

133 return u[1:-1, 1:-1] + alpha * lap 

134 

135 return TileFlagship( 

136 region=check_tier(region, "tile"), 

137 naive=naive, 

138 inputs={"u": U}, 

139 oracle=oracle, 

140 )

143def flash_tile(T: int = 6, E: int = 3, OD: int = 2, SI: int = 2, seed: int = 7) -> TileFlagship: 

144 """Flash attention as a tile body (K-D): the online-softmax state 

145 (m, den, o) carried through a fold over the s-tiles, K/V slices STAGED 

146 to "shared" per tile. Two spelling laws worth reading: 

147 

148 - the causal mask needs ABSOLUTE s coordinates inside the step, and a 

149 step has no tile coordinate — so the mask is built OUTSIDE as a free 

150 closed form (iota comparisons over the (t, s) lattice), split along 

151 s, and rides in as one more element source; 

152 - q carries no scan dim, and fold binders are exactly states+elements — 

153 so q rides in as a stride-0 repeat over "so": a broadcast element, 

154 zero bytes, sliced back to itself each iteration. 

155 

156 The final normalization needs TWO finals (o and den); the fold returns 

157 one state, so two fold nodes SHARE the step region (the adjoint 

158 machinery's re-out precedent, 260) and the division happens after. 

159 The twin materializes the softmax — same denotation, different program 

160 (flash's whole point); the deviation is the online-softmax lemma, 

161 gated by the declared ``flash.online-softmax`` license.""" 

162 SO = T // SI 

163 rng = np.random.default_rng(seed) 

164 Q = Tensor.from_numpy(rng.standard_normal((T, E)), ("t", "e")) 

165 Kv = Tensor.from_numpy(rng.standard_normal((T, E)), ("s", "e")) 

166 V = Tensor.from_numpy(rng.standard_normal((T, OD)), ("s", "o")) 

167 

168 ops = {**CORE_OPS, **TL_OPS} 

169 

170 def scores(bld, q, k): 

171 # sc[t, si] = sum_e q[t, e] * k[si, e] 

172 prod = bld.emit("tl.pointwise", bld.emit("tl.repeat_like", q, k), bld.emit("tl.repeat_like", k, q), f="mul") 

173 return bld.emit("tl.reduce", prod, dims=("e",), f="sum") 

174 

175 b = Builder(ops) 

176 pq = b.param(0, tensor_type_of_layout(Q.layout)) 

177 pk = b.param(1, tensor_type_of_layout(Kv.layout)) 

178 pv = b.param(2, tensor_type_of_layout(V.layout)) 

179 

180 # the mask, as a free closed form on the (t, s) lattice: s <= t 

181 ts = b.emit("tl.repeat", b.emit("tl.select", pq, coords=(("e", 0),)), name="s", extent=T) 

182 mask = b.emit("tl.pointwise", b.emit("tl.iota", ts, name="s"), b.emit("tl.iota", ts, name="t"), f="le") 

183 

184 # element sources over the scan dim: K/V/mask split at s, q broadcast 

185 kt = b.emit("tl.split", pk, name="s", parts=(("so", SO), ("si", SI))) 

186 vt = b.emit("tl.split", pv, name="s", parts=(("so", SO), ("si", SI))) 

187 mt = b.emit("tl.split", mask, name="s", parts=(("so", SO), ("si", SI))) 

188 qr = b.emit("tl.repeat", pq, name="so", extent=SO) 

189 

190 m0 = b.emit("tl.const", value=-1e30, dims=(("t", T),)) # flashsm's identity state 

191 den0 = b.emit("tl.const", value=0.0, dims=(("t", T),)) 

192 o0 = b.emit("tl.const", value=0.0, dims=(("t", T), ("o", OD))) 

193 

194 # the step: (m, den, o | q, k-tile, v-tile, mask-tile) -> next state 

195 sb = Builder(ops) 

196 p_m = sb.param(0, m0.type) 

197 p_den = sb.param(1, den0.type) 

198 p_o = sb.param(2, o0.type) 

199 p_q = sb.param(3, _minus_dim(qr.type, "so")) 

200 p_k = sb.param(4, _minus_dim(kt.type, "so")) 

201 p_v = sb.param(5, _minus_dim(vt.type, "so")) 

202 p_mk = sb.param(6, _minus_dim(mt.type, "so")) 

203 ks = sb.emit("tl.stage", p_k, level="shared") 

204 vs = sb.emit("tl.stage", p_v, level="shared") 

205 neg = sb.emit("core.const", type=f64, value=-1e9) 

206 sm = sb.emit("tl.pointwise", p_mk, scores(sb, p_q, ks), neg, f="where") 

207 m_new = sb.emit("tl.pointwise", p_m, sb.emit("tl.reduce", sm, dims=("si",), f="max"), f="maximum") 

208 alpha = sb.emit("tl.pointwise", sb.emit("tl.pointwise", p_m, m_new, f="sub"), f="exp") 

209 p_w = sb.emit("tl.pointwise", sb.emit("tl.pointwise", sm, sb.emit("tl.repeat_like", m_new, sm), f="sub"), f="exp") 

210 den_new = sb.emit( 

211 "tl.pointwise", 

212 sb.emit("tl.pointwise", p_den, alpha, f="mul"), 

213 sb.emit("tl.reduce", p_w, dims=("si",), f="sum"), 

214 f="add", 

215 ) 

216 pv_prod = sb.emit( 

217 "tl.pointwise", sb.emit("tl.repeat_like", p_w, vs), sb.emit("tl.repeat_like", vs, p_w), f="mul" 

218 ) 

219 o_new = sb.emit( 

220 "tl.pointwise", 

221 sb.emit("tl.pointwise", p_o, sb.emit("tl.repeat_like", alpha, p_o), f="mul"), 

222 sb.emit("tl.reduce", pv_prod, dims=("si",), f="sum"), 

223 f="add", 

224 ) 

225 step = Region( 

226 params=(p_m, p_den, p_o, p_q, p_k, p_v, p_mk), 

227 body=(sb.emit("core.yield", sb.emit("core.tuple", m_new, den_new, o_new)),), 

228 ) 

229 

230 # two finals from one loop: the fold nodes SHARE the step region 

231 fold_args = (m0, den0, o0, qr, kt, vt, mt) 

232 fold_kw = dict(dim="so", state=("m", "den", "o"), element=("q", "k", "v", "mask")) 

233 f_o = b.emit("tl.fold", *fold_args, regions=(step,), out=("final", 2), **fold_kw) 

234 f_den = b.emit("tl.fold", *fold_args, regions=(step,), out=("final", 1), **fold_kw) 

235 out = b.emit("tl.pointwise", f_o, b.emit("tl.repeat_like", f_den, f_o), f="div") 

236 region = Region(params=(pq, pk, pv), body=(b.emit("core.yield", out),)) 

237 

238 # the twin: materialized causal softmax — same denotation, different program 

239 nb = Builder(ops) 

240 nq = nb.param(0, tensor_type_of_layout(Q.layout)) 

241 nk = nb.param(1, tensor_type_of_layout(Kv.layout)) 

242 nv = nb.param(2, tensor_type_of_layout(V.layout)) 

243 sc = scores(nb, nq, nk) # (t, s) 

244 nmask = nb.emit("tl.pointwise", nb.emit("tl.iota", sc, name="s"), nb.emit("tl.iota", sc, name="t"), f="le") 

245 nsm = nb.emit("tl.pointwise", nmask, sc, nb.emit("core.const", type=f64, value=-1e9), f="where") 

246 nmax = nb.emit("tl.reduce", nsm, dims=("s",), f="max") 

247 npw = nb.emit("tl.pointwise", nb.emit("tl.pointwise", nsm, nb.emit("tl.repeat_like", nmax, nsm), f="sub"), f="exp") 

248 nden = nb.emit("tl.reduce", npw, dims=("s",), f="sum") 

249 npr = nb.emit("tl.pointwise", npw, nb.emit("tl.repeat_like", nden, npw), f="div") 

250 nprod = nb.emit("tl.pointwise", nb.emit("tl.repeat_like", npr, nv), nb.emit("tl.repeat_like", nv, npr), f="mul") 

251 nout = nb.emit("tl.reduce", nprod, dims=("s",), f="sum") 

252 naive = Region(params=(nq, nk, nv), body=(nb.emit("core.yield", nout),)) 

253 

254 def oracle(inp): 

255 sc = inp["q"] @ inp["k"].T 

256 tril = np.tril(np.ones((T, T), dtype=bool)) 

257 return np_softmax(np.where(tril, sc, -1e9), axis=1) @ inp["v"] 

258 

259 return TileFlagship( 

260 region=check_tier(region, "tile"), 

261 naive=naive, 

262 inputs={"q": Q, "k": Kv, "v": V}, 

263 oracle=oracle, 

264 )
