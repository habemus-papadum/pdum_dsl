1"""L1 — the peak-memory simulator: measure first (LEVELS.md, REPRESENTATIONS.md). 

2 

3Walks a schedule of the program tracking live bytes. What makes this exact 

4rather than heuristic is the white-box representation: 

5 

6- LAYOUT OPS ARE ALIASES: every layout op (slice/shift/repeat/pad/window/ 

7 merge/...) is a zero-byte view; a view keeps its ROOT allocation alive 

8 until the last use of ANY alias. 

9- CLOSED FORMS ARE FREE: iota and const occupy nothing (FunctionalBuffers 

10 and stride-0 broadcasts) — masks, positions, and broadcast scalars drop 

11 out of the budget structurally, exactly the "some tensors are free to 

12 save" claim. 

13- ALLOCATIONS are the compute ops (pointwise/reduce/scan/fold/materialize) 

14 and inputs; sizes come from layout shadows. 

15 

16The SCHEDULE is a separate, optimizable object (the L1 thesis): pass 

17`order` — any topological order of the same instructions — and the peak 

18moves; minimizing it is the pebbling problem the later passes attack. 

19 

20Model simplifications (documented, all conservative-or-neutral): 

21- uniform 8-byte itemsize (the shadow convention; dtype-exact sizes later); 

22- numpy's internal temporaries are ignored (reference-layer artifact); 

23- fold's transient = new-carry bytes + a recursive simulation of one step 

24 (step inputs alias outer storage but are counted — an upper bound); 

25- guarded shadows count their box (fills are reads, not bytes); 

26- a value with no later use frees immediately after its instruction; 

27- inputs are resident for the whole run unless free_inputs=True. 

28""" 

29 

30from __future__ import annotations 

31 

32from dataclasses import dataclass 

33from typing import Mapping

35# The DECLARED interior itemsize (200 §4): the reference oracle computes in 

36# f64 — a declared property, never semantics. L2's encoding assignment 

37# replaces this for materialized intermediates; BOUNDARY values take their 

38# byte truth from the inputs themselves (a Tensor's dtype, a Descriptor's 

39# encoding) — descriptor-fed, byte-exact, never the old uniform convention. 

40_INTERIOR_ITEM = 8 

41_FREE = frozenset({"iota", "const", "random"}) 

42 

43 

44def _boundary_bytes(value, layout, local: bool) -> int | None: 

45 """Byte truth for an INPUT: Descriptor -> encoding-fed; Tensor -> its 

46 dtype's itemsize; a bare Layout has no boundary fact (None).""" 

47 enc = getattr(getattr(value, "encoding", None), "nbytes", None) 

48 if enc is not None: 

49 return value.encoding.nbytes(_numel(value.layout, local)) 

50 itemsize = getattr(getattr(value, "dtype", None), "itemsize", None) 

51 if itemsize is not None: 

52 return _numel(layout, local) * itemsize 

53 return None 

54 

55 

56def _numel(layout, local: bool = False) -> int: 

57 n = 1 

58 for d in layout.dims: 

59 if local and d.level is not None: 

60 continue # bound dims index devices, not this device's memory 

61 n *= d.size 

62 return n

65@dataclass(frozen=True) 

66class MemoryReport: 

67 peak_bytes: int 

68 peak_at: str # var whose instruction hits the high-water mark 

69 live_at_peak: tuple # ((root_var_or_label, bytes), ...) at the peak 

70 timeline: tuple # (var, live_bytes_after_instr) per scheduled instr 

71 alloc_bytes: Mapping[str, int] # root var -> allocated bytes 

72 input_bytes: int

74 

75def peak_memory( 

76 region, input_layouts: dict | None = None, order=None, free_inputs: bool = False, local: bool = False, *, names=None 

77) -> MemoryReport: 

78 """With local=True, machine-bound dims (L3, PLACEMENT.md) count as 

79 device indices rather than memory: sizes become per-device shard bytes. 

80 Layouts ride the region nodes' shadows; ``input_layouts`` contributes 

81 boundary byte facts only (Tensors/Descriptors by name); ``order`` is a 

82 name sequence over the naming law's assignment — pass ``names``.""" 

83 return _region_peak_memory(region, input_layouts or {}, order, free_inputs, local, names) 

84 

85 

86def _region_peak_memory(region, inputs, order, free_inputs, local, names) -> MemoryReport: 

87 """The Region face — the same live-byte walk over dialect nodes. The 

88 default schedule is params-then-walk-order (the export order, so the 

89 Program face's trajectory is reproduced exactly); consts are deferred 

90 scalars (zero bytes, never scheduled); yields and the yielded tuple are 

91 plumbing, never uses — an unconsumed output dies at its own node, 

92 matching the Program face.""" 

93 from .dialect import LAYOUT_OP_NAMES, region_names, walk_region # lazy: dialect's consumers import this module 

94 

95 if names is None: 

96 raise ValueError( 

97 "region peak_memory joins on names — pass names=region_names(region, param_names) " 

98 "(LiftedStep.names / Assemblage.names carry it)" 

99 ) 

100 param_ids = {id(p) for p in region.params} 

101 yielded = region.body[-1].args[0] if region.body else None 

102 nodes = list(region.params) 

103 seen: set[int] = set(param_ids) 

104 for n in walk_region(region): 

105 if id(n) in seen or n.op in ("core.yield", "core.const") or (n.op == "core.tuple" and n is yielded): 

106 continue 

107 seen.add(id(n)) 

108 nodes.append(n) 

109 

110 if order is not None: 

111 by_name = {names[id(n)]: n for n in nodes} 

112 if sorted(order) != sorted(by_name): 

113 raise ValueError("order must be a permutation of the program's vars") 

114 placed: set[int] = set() 

115 reordered = [] 

116 for v in order: 

117 n = by_name[v] 

118 for a in n.args: 

119 if a.op != "core.const" and id(a) in seen and id(a) not in placed: 

120 raise ValueError(f"order is not topological: {v!r} runs before its operand {names[id(a)]!r}") 

121 placed.add(id(n)) 

122 reordered.append(n) 

123 nodes = reordered 

124 

125 root: dict[int, int | None] = {} 

126 size: dict[int, int] = {} 

127 for n in nodes: 

128 if id(n) in param_ids: 

129 root[id(n)] = id(n) 

130 size[id(n)] = _numel(n.type.layout, local) * _INTERIOR_ITEM 

131 b = _boundary_bytes(inputs.get(names[id(n)]), n.type.layout, local) 

132 if b is not None: 

133 size[id(n)] = b 

134 continue 

135 if not n.op.startswith("tl."): 

136 raise KeyError(f"no peak_memory rule for op {n.op!r}") 

137 base = n.op[3:] 

138 if base in LAYOUT_OP_NAMES or base in ("with_value_units", "repeat_like"): 

139 root[id(n)] = root[id(n.args[0])] 

140 elif base in _FREE: 

141 root[id(n)] = None 

142 else: 

143 root[id(n)] = id(n) 

144 size[id(n)] = _numel(n.type.layout, local) * _INTERIOR_ITEM 

145 

146 def transient(n) -> int: 

147 if n.op != "tl.fold": 

148 return 0 

149 p = dict(n.attrs) 

150 state, element = tuple(p["state"]), tuple(p["element"]) 

151 k = len(state) 

152 carry_bytes = sum(size.get(id(a), _numel(a.type.layout, local) * _INTERIOR_ITEM) for a in n.args[:k]) 

153 step = n.regions[0] 

154 sub = _region_peak_memory(step, {}, None, True, local, region_names(step, state + element)) 

155 return carry_bytes + sub.peak_bytes 

156 

157 last: dict[int, int] = {} 

158 for i, n in enumerate(nodes): 

159 if id(n) not in param_ids: 

160 for a in n.args: 

161 r = root.get(id(a)) 

162 if r is not None: 

163 last[r] = i 

164 r = root[id(n)] 

165 if r is not None: 

166 last.setdefault(r, i) # dead values free immediately 

167 

168 live = 0 

169 live_set: dict[int, int] = {} 

170 peak, peak_at, at_peak = 0, "", () 

171 timeline = [] 

172 for i, n in enumerate(nodes): 

173 alloc = size[id(n)] if root[id(n)] == id(n) else 0 

174 tr = transient(n) 

175 if live + alloc + tr > peak: 

176 peak, peak_at = live + alloc + tr, names[id(n)] 

177 snap = {names[r]: b for r, b in live_set.items()} 

178 if alloc: 

179 snap[names[id(n)]] = alloc 

180 if tr: 

181 snap["(fold transient)"] = tr 

182 at_peak = tuple(sorted(snap.items())) 

183 if alloc: 

184 live += alloc 

185 live_set[id(n)] = alloc 

186 for r, li in list(last.items()): 

187 if li == i and r in live_set and (free_inputs or r not in param_ids): 

188 live -= live_set.pop(r) 

189 timeline.append((names[id(n)], live)) 

190 return MemoryReport( 

191 peak_bytes=peak, 

192 peak_at=peak_at, 

193 live_at_peak=at_peak, 

194 timeline=tuple(timeline), 

195 alloc_bytes={names[r]: b for r, b in size.items()}, 

196 input_bytes=sum(size[id(p)] for p in region.params), 

197 )
