1"""Coordinate charts (step 3): exact physical labels for lattice dims. 

2 

3A Chart is an affine relabeling of a dim's integer lattice: 

4 

5 phys(i) = origin + i * step (origin, step exact Quantities) 

6 

7Today's plain-integer indexing is the degenerate chart (origin 0, step 1, 

8dimensionless) left implicit; attaching a chart changes no address or guard 

9arithmetic — the lattice keeps full citizenship, charts are the API surface. 

10 

11`kind` records the affine/vector distinction: positions have an origin 

12(stage coordinates, timestamps); displacements are origin-free differences 

13(kernel taps, block-local offsets). 

14 

15`axis` names the physical axis a dim lives on. Several dims may share one 

16axis — split parts, window/stencil kernel dims — and the invariant is: 

17 

18 physical position on axis A = sum of the chart labels of all dims 

19 tagged A (at most one of which is a position; the rest displacements). 

20 

21Ops maintain the invariant mechanically: split tags its parts with the 

22parent's axis, kernels get the anchor's axis, and select folds a removed 

23dim's label into a remaining sibling on the same axis (promoting a 

24displacement to a position when the position itself was selected). This is 

25what makes "physical labels stay glued to the data" a theorem across ops 

26rather than a per-op convention. 

27 

28Membership is exact-only: `lattice()` raises off-lattice; `snap()` is the 

29explicit, deliberate way to round onto the lattice. 

30""" 

31 

32from __future__ import annotations 

33 

34import math 

35from dataclasses import dataclass 

36from fractions import Fraction

38from .units import Quantity 

39from .units import q as _q

41KINDS = ("position", "displacement")

44def _as_quantity(value) -> Quantity: 

45 if isinstance(value, Quantity): 

46 return value 

47 return _q(value) # str/int/Fraction; floats rejected inside 

48 

49 

50def _fr_gcd(a: Fraction, b: Fraction) -> Fraction: 

51 return Fraction( 

52 math.gcd(a.numerator * b.denominator, b.numerator * a.denominator), 

53 a.denominator * b.denominator, 

54 ) 

55 

56 

57@dataclass(frozen=True) 

58class Chart: 

59 origin: Quantity 

60 step: Quantity 

61 kind: str = "position" 

62 axis: str | None = None # name of the physical axis this dim labels 

63 

64 def __post_init__(self) -> None: 

65 object.__setattr__(self, "origin", _as_quantity(self.origin)) 

66 object.__setattr__(self, "step", _as_quantity(self.step)) 

67 if self.origin.dims != self.step.dims: 

68 raise ValueError(f"origin {self.origin!r} and step {self.step!r} have different dimensions") 

69 if not self.step: 

70 raise ValueError("chart step must be nonzero") 

71 if self.kind not in KINDS: 

72 raise ValueError(f"kind must be one of {KINDS}, got {self.kind!r}") 

73 

74 def phys(self, i: int) -> Quantity: 

75 return self.origin + i * self.step 

76 

77 def ratio(self, value) -> Fraction: 

78 """(value - origin) / step as an exact rational lattice position.""" 

79 return (_as_quantity(value) - self.origin) / self.step 

80 

81 def lattice(self, value) -> int: 

82 """Exact-only membership: raises if `value` is off-lattice.""" 

83 r = self.ratio(value) 

84 if r.denominator != 1: 

85 raise ValueError( 

86 f"{_as_quantity(value)!r} is off-lattice for {self!r} " 

87 f"(lattice position {r}); use snap() to round deliberately" 

88 ) 

89 return int(r) 

90 

91 def snap(self, value, mode: str = "nearest") -> int: 

92 """Deliberate rounding onto the lattice.""" 

93 r = self.ratio(value) 

94 if mode == "floor": 

95 return math.floor(r) 

96 if mode == "ceil": 

97 return math.ceil(r) 

98 if mode == "nearest": 

99 return round(r) # ties to even, like round() 

100 raise ValueError(f"mode must be floor/ceil/nearest, got {mode!r}") 

101 

102 def commensurable(self, other: "Chart") -> bool: 

103 """Can the two lattices interoperate pointwise (no resampling)? 

104 True iff same dimensions and the origins differ by an integer 

105 multiple of the gcd of the steps.""" 

106 if self.step.dims != other.step.dims: 

107 return False 

108 g = _fr_gcd(abs(self.step.base), abs(other.step.base)) 

109 return ((self.origin.base - other.origin.base) / g).denominator == 1 

110 

111 def __repr__(self) -> str: 

112 k = "pos" if self.kind == "position" else "disp" 

113 ax = f"{self.axis}: " if self.axis is not None else "" 

114 return f"{k}[{ax}{self.origin!r} step {self.step!r}]" 

115 

116 

117def chart(origin, step, kind: str = "position", axis: str | None = None) -> Chart: 

118 """Convenience constructor coercing strings/ints/Fractions exactly: 

119 chart("-1.5 um", "0.25 um").""" 

120 return Chart(_as_quantity(origin), _as_quantity(step), kind, axis) 

121

123def characteristic(rate, label_along: str): 

124 """Diagonal chart combinator for a rate-related pair of axes (x = rate·t, 

125 e.g. a wave characteristic): validates that the lattice is EXACTLY 

126 characteristic — step(label_along) == rate * step(other), the CFL-style 

127 commensurability condition — and labels z along `label_along` (a dim 

128 name). All the physics lives here, named and testable; `diagonal` itself 

129 never guesses. 

130 

131 xi = t2d.diagonal(("x", "t"), "xi", 

132 chart=characteristic(q("2 um/ms"), "x")) 

133 """ 

134 rate = _as_quantity(rate) 

135 

136 def combine(dims) -> Chart: 

137 if len(dims) != 2: 

138 raise ValueError("characteristic() combines exactly two dims") 

139 by = {d.name: d for d in dims} 

140 if label_along not in by: 

141 raise KeyError(f"{label_along!r} is not one of {sorted(by)}") 

142 d_lab = by.pop(label_along) 

143 (d_other,) = by.values() 

144 if d_lab.chart is None or d_other.chart is None: 

145 raise TypeError("characteristic() needs charts on both dims") 

146 expect = rate * d_other.chart.step 

147 if d_lab.chart.step != expect: 

148 raise ValueError( 

149 f"lattice is not characteristic for rate {rate!r}: " 

150 f"step({d_lab.name}) = {d_lab.chart.step!r} but rate * " 

151 f"step({d_other.name}) = {expect!r}" 

152 ) 

153 return d_lab.chart 

154 

155 return combine
